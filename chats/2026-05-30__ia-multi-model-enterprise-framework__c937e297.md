function recordFeedback(skillId, score) {
  const skill = state.skills.find(s => s.id === skillId);
  if (!skill) return false;
  // Actualización exponencial suave (0.8 + 0.2*score)
  const oldRate = skill.successRate;
  skill.successRate = Math.min(0.99, Math.max(0.3, oldRate * 0.8 + score * 0.2));
  skill.version += 1;
  state.evolution.feedbackLog.push({ skillId, score, ts: Date.now(), oldRate, newRate: skill.successRate });
  
  // Guardar versión histórica (para seguimiento)
  if (!state.evolution.versionHistory[skillId]) state.evolution.versionHistory[skillId] = [];
  state.evolution.versionHistory[skillId].push({ version: skill.version, successRate: skill.successRate, timestamp: Date.now() });
  
  try { localStorage.setItem('belentani_evolution', JSON.stringify({ feedbackLog: state.evolution.feedbackLog, versionHistory: state.evolution.versionHistory })); } catch(e) {}
  return true;
}

// ========== ROUTING LOGIC ==========
function detectTaskType(prompt) {
  const p = prompt.toLowerCase();
  if (p.includes('code') || p.includes('function') || p.includes('api')) return 'code_generation';
  if (p.includes('story') || p.includes('creative') || p.includes('write')) return 'creative_writing';
  if (p.includes('analyze') || p.includes('data') || p.includes('chart')) return 'data_analysis';
  if (p.includes('chinese') || p.includes('mandarin') || p.includes('中文')) return 'multilingual_zh';
  if (p.includes('long') || p.includes('document') || p.includes('100k')) return 'long_context_100k';
  if (p.includes('cheap') || p.includes('cost') || p.includes('budget')) return 'cost_optimized';
  if (p.includes('fast') || p.includes('quick') || p.includes('latency')) return 'low_latency';
  return 'fallback_safe';
}

function selectProvider(prompt, priority = 'balanced') {
  const task = detectTaskType(prompt);
  const chain = state.routingRules[task] || state.routingRules.fallback_safe;
  
  // Simple priority weighting
  if (priority === 'cost') chain.sort((a,b) => state.providers[a].cost.output - state.providers[b].cost.output);
  if (priority === 'latency') chain.sort((a,b) => state.providers[a].latency - state.providers[b].latency);
  
  // Return first online provider in chain
  for (const pid of chain) {
    if (state.providers[pid].status === 'online') return { provider: state.providers[pid], taskType: task, chain };
  }
  return { provider: state.providers.openrouter, taskType: task, chain }; // ultimate fallback
}

# core/router.py
def select_provider(prompt: str, priority: str = "balanced") -> dict:
    task = detect_task_type(prompt)  # code, seo, video, etc.
    chain = ROUTING_RULES.get(task, ROUTING_RULES["fallback_safe"])
    
    if priority == "cost":
        chain.sort(key=lambda p: PROVIDERS[p]["cost_per_output"])
    elif priority == "latency":
        chain.sort(key=lambda p: PROVIDERS[p]["avg_latency_ms"])
    
    # Devuelve primer proveedor online en cadena
    for pid in chain:
        if is_provider_online(pid):
            return {"provider": PROVIDERS[pid], "task": task, "chain": chain}
    
    return {"provider": PROVIDERS["openrouter"], "task": task}  # Último fallback

    $audioScript = @'
#!/usr/bin/env python3
"""Belentani Audio Processor - Unifica todas las herramientas de audio"""
import os, sys, subprocess
from pathlib import Path

class BelentaniAudio:
    def __init__(self, project_dir: str):
        self.project_dir = Path(project_dir)
        self.audio_dir = self.project_dir / "audio"
        self.audio_dir.mkdir(exist_ok=True)
    
    def separate_stems(self, input_file: str, output_dir: str = None):
        """Separa vocales e instrumental con Demucs"""
        output = output_dir or str(self.audio_dir / "stems")
        cmd = ["demucs", "--two-stems=vocals", input_file, "-o", output]
        subprocess.run(cmd, check=True)
        return {"vocals": f"{output}/vocals.wav", "instrumental": f"{output}/no_vocals.wav"}
    
    def clone_voice(self, ref_audio: str, text: str, output: str):
        """Clona voz con F5-TTS (10s de referencia)"""
        from f5_tts.infer import infer
        result = infer(ref_audio=ref_audio, gen_text=text, remove_silence=True)
        result.save(output)
        return output
    
    def generate_music(self, prompt: str, duration: int = 30, output: str = None):
        """Genera instrumental con MusicGen"""
        from audiocraft.models import MusicGen
        from audiocraft.data.audio import audio_write
        
        model = MusicGen.get_pretrained('facebook/musicgen-small')
        model.set_generation_params(duration=duration)
        descriptions = [prompt]
        wav = model.generate(descriptions)
        
        output_path = output or str(self.audio_dir / "generated.wav")
        audio_write(output_path.replace('.wav', ''), wav[0].cpu(), model.sample_rate, strategy="loudness")
        return f"{output_path}.wav"
    
    def master(self, input_file: str, reference: str = None, output: str = None):
        """Masteriza con Matchering (opcional: con referencia profesional)"""
        import matchering as mg
        output_path = output or input_file.replace('.wav', '_mastered.wav')
        mg.process(target=input_file, reference=reference, output_file=output_path)
        return output_path
    
    def clean_audio(self, input_file: str, output: str = None):
        """Limpia ruido, clicks, pops con ClearVoice"""
        from clearvoice import ClearVoice
        cv = ClearVoice()
        output_path = output or input_file.replace('.wav', '_clean.wav')
        cv.process(input_file, output_path)
        return output_path

# Uso rápido:
# audio = BelentaniAudio("C:/Users/Pedro/BelentaniOmnipotent")
# stems = audio.separate_stems("cancion.mp3")
# audio.clone_voice("pedro_ref.wav", "Nueva letra", "cover_vocal.wav")
# audio.master("mix.wav", reference="pro_track.wav", output="final_mastered.wav")
'@

from typing import Optional, List

class BelentaniRouter:
    """Router inteligente con Graphify integration"""
    
    def __init__(self, config_path: str):
        with open(config_path, 'r', encoding='utf-8') as f:
            self.config = json.load(f)
        self.api_key = os.getenv('OPENROUTER_API_KEY', '')

        if not self.graphify_enabled or not self.graph_path:

        # Intentar encontrar respuesta en el grafo primero

                return f"🔍 Encontrado en grafo: {node.get('label')} — {node.get('description', '')}"
        
        return None
    
    def select_model(self, task_type: str) -> str:
        """Selecciona modelo óptimo + verifica grafo primero"""
        # Intentar responder desde el grafo primero (ahorra tokens)
        if task_type in ['graph_query', 'knowledge_traversal']:
            graph_answer = self.query_graph("dummy")  # Placeholder
            if graph_answer:
                return None  # No need to call LLM
        
        rules = self.config['routing_rules']
        candidates = rules.get(task_type, rules['default'])
        return candidates[0]
    
    def call(self, prompt: str, task_type: str = "default", use_graph: bool = True, **kwargs):
        """Llama a OpenRouter con optimización Graphify"""
        # Paso 1: Intentar responder desde el grafo (si está habilitado)
        if use_graph and self.graphify_enabled:
            graph_answer = self.query_graph(prompt)
            if graph_answer:
                return {"from_graph": True, "answer": graph_answer, "tokens_saved": "~2000"}
        
        # Paso 2: Llamada normal a LLM
        model = self.select_model(task_type)
        if not model:
            return {"error": "No model selected"}
        
        headers = {
            "Authorization": f"Bearer {self.api_key}",
            "HTTP-Referer": "https://belentani.com",
            "X-Title": "Belentani Omnipotent v3"
        }
        payload = {
            "model": model,
            "messages": [{"role": "user", "content": prompt}],
            "max_tokens": kwargs.get('max_tokens', 8000),
            "temperature": kwargs.get('temperature', 0.7)
        }
        
        response = requests.post(f"{self.base_url}/chat/completions", headers=headers, json=payload)
        return response.json()
