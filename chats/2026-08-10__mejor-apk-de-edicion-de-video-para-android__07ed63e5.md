data class RemoteConfig(
    val config_version: Long,
    val min_app_version: String,
    val force_refresh: Boolean,
    val device_profiles: Map<String, DeviceProfile>,
    val scene_profiles: Map<String, SceneProfile>,

class ConfigRepository(
    private val api: ConfigApi,
    private val store: ConfigStore
) {
    suspend fun forceRefreshIfNeeded(): RemoteConfig {
        val local = store.load()
        val remote = api.getLatestConfig()

        if (remote.config_version > local?.config_version || remote.force_refresh) {
            verifyOrThrow(remote)
            store.save(remote)
            return remote
        }

        return local
    }

    private fun verifyOrThrow(config: RemoteConfig) {
        if (!SignatureVerifier.isValid(config)) {
            throw SecurityException("Config inválida")
        }
    }
}

# 6. Algoritmos clave para fotografía nocturna

## 6.1 Estrategia base

La fotografía en oscuridad no debe depender solo de subir ISO. Debe usar:

1. Captura múltiple.
2. Alineación.
3. Reducción de ruido temporal.
4. Eliminación de objetos fantasma.
5. Realce de sombras sin amplificar ruido.
6. Tone mapping suave.
7. Conservación de color.

## 6.2 Pseudocódigo del modo noche

```python
def night_mode(frames, config):
    # frames: lista de capturas RAW/YUV
    aligned = align_frames(frames)
    motion_mask = detect_motion(aligned)

    merged = merge_frames(
        aligned,
        method="temporal_median_weighted",
        motion_rejection=True,
        weight_by_noise=True
    )

    raw_linear = demosaic_and_linearize(merged)
    denoised = ai_denoise(raw_linear, config.denoise_strength)

    shadows = enhance_low_light(
        denoised,
        model=config.low_light_model,
        strength=config.shadow_boost
    )

    tone_mapped = tone_map(
        shadows,
        method=config.tone_mapping,
        highlight_preserve=0.85
    )

    color_corrected = adaptive_white_balance(tone_mapped)
    final = sharpen_edge_aware(color_corrected, amount=config.sharpen)

    return final
```

# 7. HDR+ y rango dinámico

## 7.1 Método recomendado

Para móviles, la fusión por exposición suele ser más estable que Debevec tradicional.

```python
def hdr_plus(frames):
    aligned = align_frames(frames)
    ghosts = detect_ghosts(aligned)
    fused = exposure_fusion(
        aligned,
        contrast_weight=0.5,
        saturation_weight=0.3,
        well_exposedness_weight=0.7
    )
    fused = remove_ghosts(fused, ghosts)
    fused = local_tone_map(fused, strength=0.6)
    return fused
```

## 7.2 Parámetros

- Bracketing: -2 EV, 0 EV, +2 EV.

- Fusión: Mertens exposure fusion.
- Ghost removal obligatorio en escenas con personas/vegetación.
- Local tone mapping con refuerzo de detalle en sombras y protección de altas luces.

# 12. Pipeline de imagen en GPU

Para rendimiento, el pipeline debe usar shaders.

## 12.1 Orden recomendado

```text
YUV → RGB
  ↓
Linearize
  ↓
Denoise
  ↓
HDR merge
  ↓
Tone map
  ↓
Color matrix
  ↓
Saturation/vibrance
  ↓
Sharpen
  ↓
Output
```

## 12.2 Shader simple de tono y saturación

```glsl
#version 320 es
precision mediump float;

uniform sampler2D inputTexture;
uniform float saturation;
uniform float contrast;
uniform float brightness;

in vec2 uv;
out vec4 outColor;

void main() {
    vec4 color = texture(inputTexture, uv);

    color.rgb = (color.rgb - 0.5) * contrast + 0.5 + brightness;

    float luma = dot(color.rgb, vec3(0.299, 0.587, 0.114));
    vec3 saturated = mix(vec3(luma), color.rgb, saturation);

    outColor = vec4(clamp(saturated, 0.0, 1.0), color.a);
}
```

def merge_frames(frames):

# 21. Decisión técnica recomendada

Si quieres el mejor resultado real sin piratería ni clonación propietaria, construye:

- **Captura:** Camera2 + burst multi-frame.
- **Noche:** alineación + fusión temporal + denoise IA + tone mapping suave.
- **HDR:** exposure fusion + ghost removal.
- **IA:** modelos ligeros cuantizados en TFLite/ONNX.
- **Config:** backend con remote config firmado y forzado por versión.
- **Editor:** pipeline no destructivo con operaciones JSON.
- **Escalabilidad:** backend opcional para procesamiento pesado.

Con eso obtienes una app full-stack potente, legal, actualizable y con capacidad real de mejorar mediante configuración remota y modelos IA.
