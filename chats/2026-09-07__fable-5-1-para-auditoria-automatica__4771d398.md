            dirs[:] = [d for d in dirs if d not in {'node_modules', '.git', '__pycache__', 'venv', 'dist', 'build'}]
            
            root_path = Path(root)
            
            # Detectar proyectos (tienen package.json, requirements.txt, .git, etc.)
            is_project = any([
                (root_path / 'package.json').exists(),
                (root_path / 'requirements.txt').exists(),
                (root_path / 'pyproject.toml').exists(),
                (root_path / '.git').exists(),
                (root_path / 'pom.xml').exists(),
                (root_path / 'Cargo.toml').exists()
            ])
            
            if is_project:
                self._analyze_project(root_path)
    
    def _analyze_project(self, project_path: Path):
        """Análisis profundo de proyecto (estilo Netflix code quality)"""
        try:
            # Calcular tamaño y archivos
            total_size = 0
            file_count = 0
            languages = defaultdict(int)
            
            for file_path in project_path.rglob('*'):
                if file_path.is_file():
                    total_size += file_path.stat().st_size
                    file_count += 1
                    
                    # Detectar lenguaje
                    ext = file_path.suffix.lower()
                    if ext == '.py':
                        languages['Python'] += 1
                    elif ext in ['.js', '.ts', '.jsx', '.tsx']:
                        languages['TypeScript'] += 1
                    elif ext == '.html':
                        languages['HTML'] += 1
                    elif ext == '.css':
                        languages['CSS'] += 1
            
            # Detectar lenguaje dominante
            language = max(languages.items(), key=lambda x: x[1])[0] if languages else 'Unknown'
            
            # Metadata del proyecto
            metadata = ProjectMetadata(
                name=project_path.name,
                path=project_path,
                size_mb=total_size / (1024 * 1024),
                file_count=file_count,
                language=language,
                is_git_repo=(project_path / '.git').exists(),
                last_modified=datetime.fromtimestamp(project_path.stat().st_mtime)
            )
            
            # Auditoría de calidad
            metadata.quality_score = self._audit_quality(project_path, metadata)
            
            # Detectar problemas
            metadata.issues = self._detect_issues(project_path, metadata)
            
            self.projects[str(project_path)] = metadata
            logger.info(f"✅ Analizado: {metadata.name} ({metadata.size_mb:.1f} MB, {metadata.quality_score:.0f}% calidad)")
            
        except Exception as e:
            logger.error(f"❌ Error analizando {project_path}: {e}")
    
    def _audit_quality(self, project_path: Path, metadata: ProjectMetadata) -> float:
        """Auditoría de calidad (0-100) basada en estándares enterprise"""
        score = 100
        
        # Penalizaciones
        if not (project_path / 'README.md').exists():
            score -= 20
            metadata.issues.append("Falta README.md")
        
        if not (project_path / '.gitignore').exists():
            score -= 15
            metadata.issues.append("Falta .gitignore")
        
        # Verificar archivos de configuración
        config_files = ['package.json', 'requirements.txt', 'pyproject.toml']
        if not any((project_path / f).exists() for f in config_files):
            score -= 10
            metadata.issues.append("Falta archivo de dependencias")
        
        # Verificar tests
        test_dirs = ['tests', 'test', '__tests__']
        if not any((project_path / d).exists() for d in test_dirs):
            score -= 15
            metadata.issues.append("Falta directorio de tests")
        
        # Verificar CI/CD
        if not (project_path / '.github').exists():
            score -= 10
            metadata.issues.append("Falta configuración CI/CD")
        
        # Verificar licencia
        license_files = ['LICENSE', 'LICENSE.md', 'LICENSE.txt']
        if not any((project_path / f).exists() for f in license_files):
            score -= 5
            metadata.issues.append("Falta licencia")
        
        return max(0, score)

                if 'dependencies' in package_json:

    def detect_duplicates(self):
        """Detección de duplicados (estilo dupeguru/librephotos)"""
        logger.info("🔍 Detectando duplicados con hashing SHA-256...")
        
        file_hashes = defaultdict(list)
        
        # Hash de todos los archivos
        for project_path, metadata in self.projects.items():
            for file_path in Path(project_path).rglob('*'):
                if file_path.is_file() and file_path.stat().st_size > 0:
                    file_hash = self.calculate_hash(file_path)
                    if file_hash:
                        file_hashes[file_hash].append(file_path)
        
        # Filtrar duplicados
        for file_hash, file_paths in file_hashes.items():
            if len(file_paths) > 1:
                # Agrupar por directorio padre
                parent_dirs = defaultdict(list)
                for path in file_paths:
                    parent_dirs[path.parent].append(path)
                
                # Si hay duplicados en diferentes proyectos
                if len(parent_dirs) > 1:
                    self.duplicates[file_hash] = file_paths
                    self.report['duplicates_found'] += 1
        
        logger.info(f"📊 Duplicados encontrados: {len(self.duplicates)}")

            metadata = ProjectMetadata(
                name=project_path.name,
                path=project_path,
                size_mb=total_size / (1024 * 1024),
                file_count=file_count,
                language=language,
                is_git_repo=(project_path / '.git').exists(),
                last_modified=datetime.fromtimestamp(project_path.stat().st_mtime)
            )
            
            # Auditoría de calidad
            metadata.quality_score = self._audit_quality(project_path, metadata)
            
            # Detectar problemas
            metadata.issues = self._detect_issues(project_path, metadata)
            
            self.projects[str(project_path)] = metadata
            logger.info(f"✅ Analizado: {metadata.name} ({metadata.size_mb:.1f} MB, {metadata.quality_score:.0f}% calidad)")
            
        except Exception as e:
            logger.error(f"❌ Error analizando {project_path}: {e}")
    
    def _audit_quality(self, project_path: Path, metadata: ProjectMetadata) -> float:
        """Auditoría de calidad (0-100) basada en estándares enterprise"""
        score = 100
        
        # Penalizaciones
        if not (project_path / 'README.md').exists():
            score -= 20
            metadata.issues.append("Falta README.md")
        
        if not (project_path / '.gitignore').exists():
            score -= 15
            metadata.issues.append("Falta .gitignore")
        
        # Verificar archivos de configuración
        config_files = ['package.json', 'requirements.txt', 'pyproject.toml']
        if not any((project_path / f).exists() for f in config_files):
            score -= 10
            metadata.issues.append("Falta archivo de dependencias")
        
        # Verificar tests
        test_dirs = ['tests', 'test', '__tests__']
        if not any((project_path / d).exists() for d in test_dirs):
            score -= 15
            metadata.issues.append("Falta directorio de tests")
        
        # Verificar CI/CD
        if not (project_path / '.github').exists():
            score -= 10
            metadata.issues.append("Falta configuración CI/CD")
        
        # Verificar licencia
        license_files = ['LICENSE', 'LICENSE.md', 'LICENSE.txt']
        if not any((project_path / f).exists() for f in license_files):
            score -= 5
            metadata.issues.append("Falta licencia")
        
        return max(0, score)

    def calculate_hash(self, file_path: Path) -> str:
        """Hash SHA-256 optimizado para archivos grandes"""
        sha256_hash = hashlib.sha256()
        try:
            with open(file_path, "rb") as f:
                for byte_block in iter(lambda: f.read(8192), b""):
                    sha256_hash.update(byte_block)
            return sha256_hash.hexdigest()
        except Exception as e:
            logger.warning(f"Error hashing {file_path}: {e}")
            return ""
    
    def scan_ecosystem(self):
        """Escaneo paralelo del ecosistema (estilo Google Code Search)"""
        logger.info("🔍 Iniciando escaneo enterprise paralelo...")
        
        project_paths = []
        for root, dirs, files in os.walk(self.root):
            # Ignorar directorios de build y dependencias
            dirs[:] = [d for d in dirs if d not in {
                'node_modules', '.git', '__pycache__', 'venv', 'dist', 
                'build', '.next', 'coverage', '.cache'
            }]
            
            root_path = Path(root)
            
            # Detectar proyectos
            is_project = any([
                (root_path / 'package.json').exists(),
                (root_path / 'requirements.txt').exists(),
                (root_path / 'pyproject.toml').exists(),
                (root_path / '.git').exists(),
                (root_path / 'pom.xml').exists(),
                (root_path / 'Cargo.toml').exists(),
                (root_path / 'go.mod').exists(),
                (root_path / 'Makefile').exists()
            ])
            
            if is_project:
                project_paths.append(root_path)
        
        # Análisis paralelo
        with ThreadPoolExecutor(max_workers=self.max_workers) as executor:
            futures = {executor.submit(self._analyze_project, p): p for p in project_paths}
            for future in as_completed(futures):
                try:
                    future.result()
                except Exception as e:
                    logger.error(f"Error en análisis: {e}")
        
        logger.info(f"✅ {len(self.projects)} proyectos analizados")
    
    def _analyze_project(self, project_path: Path):
        """Análisis profundo de proyecto (estilo Netflix/Spotify)"""
        try:
            # Calcular métricas
            total_size = 0
            file_count = 0
            languages = Counter()
            tech_stack = set()
            
            for file_path in project_path.rglob('*'):
                if file_path.is_file():
                    total_size += file_path.stat().st_size
                    file_count += 1
                    
                    # Detectar lenguaje y tech stack
                    ext = file_path.suffix.lower()
                    name = file_path.name.lower()
                    
                    if ext == '.py':
                        languages['Python'] += 1
                        tech_stack.add('Python')
                    elif ext in ['.js', '.ts', '.jsx', '.tsx']:
                        languages['TypeScript'] += 1
                        tech_stack.add('TypeScript')
                    elif ext == '.html':
                        languages['HTML'] += 1
                        tech_stack.add('HTML')
                    elif ext == '.css':
                        languages['CSS'] += 1
                        tech_stack.add('CSS')
                    elif ext == '.go':
                        languages['Go'] += 1
                        tech_stack.add('Go')
                    elif ext == '.java':
                        languages['Java'] += 1
                        tech_stack.add('Java')
                    
                    # Detectar frameworks
                    if 'react' in name:
                        tech_stack.add('React')
                    if 'vue' in name:
                        tech_stack.add('Vue')
                    if 'angular' in name:
                        tech_stack.add('Angular')
                    if 'next' in name:
                        tech_stack.add('Next.js')
                    if 'django' in name:
                        tech_stack.add('Django')
                    if 'flask' in name:
                        tech_stack.add('Flask')
                    if 'fastapi' in name:
                        tech_stack.add('FastAPI')
            
            # Lenguaje dominante
            language = languages.most_common(1)[0][0] if languages else 'Unknown'
            
            # Metadata
            metadata = ProjectMetadata(
                name=project_path.name,
                path=project_path,
                size_mb=total_size / (1024 * 1024),
                file_count=file_count,
                language=language,
                is_git_repo=(project_path / '.git').exists(),
                last_modified=datetime.fromtimestamp(project_path.stat().st_mtime),
                tech_stack=list(tech_stack)
            )
            
            # Análisis avanzado
            metadata.nicho = self._detect_nicho(metadata)
            metadata.categoria = self._detect_categoria(metadata)
            metadata.quality_score = self._audit_quality(project_path, metadata)
            metadata.issues = self._detect_issues(project_path, metadata)
            metadata.security_issues = self._detect_security_issues(project_path)
            metadata.git_commits = self._count_git_commits(project_path)
            
            self.projects[str(project_path)] = metadata
            logger.info(f"✅ {metadata.name} | {metadata.nicho} | {metadata.quality_score:.0f}% | {metadata.size_mb:.1f}MB")
            
        except Exception as e:
            logger.error(f"❌ Error analizando {project_path}: {e}")

    def _audit_quality(self, project_path: Path, metadata: ProjectMetadata) -> float:
        """Auditoría de calidad enterprise (0-100)"""
        score = 100
        
        # Documentación
        if not (project_path / 'README.md').exists():
            score -= 15
            metadata.issues.append("Falta README.md")
        else:
            metadata.has_docs = True
        
        # Git
        if not (project_path / '.gitignore').exists():
            score -= 10
            metadata.issues.append("Falta .gitignore")
        
        # Dependencias
        config_files = ['package.json', 'requirements.txt', 'pyproject.toml', 'Cargo.toml', 'go.mod']
        if not any((project_path / f).exists() for f in config_files):
            score -= 10
            metadata.issues.append("Falta archivo de dependencias")
        
        # Tests
        test_dirs = ['tests', 'test', '__tests__', 'spec']
        if any((project_path / d).exists() for d in test_dirs):
            metadata.has_tests = True
        else:
            score -= 15
            metadata.issues.append("Falta directorio de tests")
        
        # CI/CD
        if (project_path / '.github' / 'workflows').exists():
            metadata.has_ci_cd = True
        else:
            score -= 10
            metadata.issues.append("Falta configuración CI/CD")
        
        # Licencia
        license_files = ['LICENSE', 'LICENSE.md', 'LICENSE.txt']
        if not any((project_path / f).exists() for f in license_files):
            score -= 5
            metadata.issues.append("Falta licencia")
        
        # Commits
        if metadata.git_commits < 10:
            score -= 5
            metadata.issues.append(f"Pocos commits ({metadata.git_commits})")
        
        return max(0, score)
    
    def _detect_issues(self, project_path: Path, metadata: ProjectMetadata) -> List[str]:
        """Detección de problemas de código"""
        issues = []
        
        # Buscar TODOs, FIXMEs, HACKs
        todo_patterns = [r'TODO', r'FIXME', r'HACK', r'XXX', r'BUG']
        todo_count = 0
        
        for file_path in project_path.rglob('*'):
            if file_path.is_file() and file_path.suffix in ['.py', '.js', '.ts', '.html']:
                try:
                    content = file_path.read_text(encoding='utf-8', errors='ignore')
                    for pattern in todo_patterns:
                        matches = len(re.findall(pattern, content, re.IGNORECASE))
                        todo_count += matches
                except:
                    pass
        
        if todo_count > 20:
            issues.append(f"Muchos TODOs/FIXMEs ({todo_count})")
        
        # Archivos grandes
        for file_path in project_path.rglob('*'):
            if file_path.is_file() and file_path.stat().st_size > 5 * 1024 * 1024:  # 5MB
                if file_path.suffix not in ['.zip', '.tar', '.gz', '.mp4', '.mp3']:
                    issues.append(f"Archivo grande: {file_path.name} ({file_path.stat().st_size / 1024 / 1024:.1f}MB)")
        
        return issues
    
    def _detect_security_issues(self, project_path: Path) -> List[str]:
        """Detección de problemas de seguridad (OWASP)"""
        issues = []
        
        # Secrets expuestos
        secret_patterns = [
            r'(?i)(password|passwd|pwd)\s*=\s*["\']([a-zA-Z0-9+/=]{8,})["\']',
            r'(?i)(secret|token|key|api_key|apikey)\s*=\s*["\']([a-zA-Z0-9+/=]{16,})["\']',
            r'(?i)(aws_access_key_id|aws_secret_access_key)\s*=\s*["\']([A-Z0-9]{20,})["\']',
            r'(?i)(private_key|ssh_key)\s*=\s*["\']([a-zA-Z0-9+/=]{32,})["\']',
            r'-----BEGIN (RSA |EC )?PRIVATE KEY-----'
        ]
        
        for file_path in project_path.rglob('*'):
            if file_path.is_file() and file_path.suffix in ['.py', '.js', '.ts', '.env', '.json']:
                try:
                    content = file_path.read_text(encoding='utf-8', errors='ignore')
                    for pattern in secret_patterns:
                        if re.search(pattern, content):
                            issues.append(f"⚠️  Secret expuesto en {file_path.name}")
                            break
                except:
                    pass
        
        # Archivos .env en git
        if (project_path / '.env').exists():
            gitignore_path = project_path / '.gitignore'
            if gitignore_path.exists():
                gitignore = gitignore_path.read_text()
                if '.env' not in gitignore:
                    issues.append("⚠️  .env no está en .gitignore")
        
        return issues
    
    def _count_git_commits(self, project_path: Path) -> int:
        """Contar commits de git"""
        if not (project_path / '.git').exists():
            return 0
        
        try:
            result = subprocess.run(
                ['git', 'rev-list', '--count', 'HEAD'],
                cwd=project_path,
                capture_output=True,
                text=True,
                timeout=5
            )
            return int(result.stdout.strip()) if result.returncode == 0 else 0
        except:
            return 0
    
    def detect_duplicates(self):
        """Detección inteligente de duplicados (estilo dupeguru)"""
        logger.info("🔍 Detectando duplicados con hashing SHA-256...")
        
        file_hashes = defaultdict(list)
        
        # Hash paralelo de archivos
        with ThreadPoolExecutor(max_workers=self.max_workers) as executor:
            futures = []
            for project_path in self.projects.values():
                for file_path in Path(project_path.path).rglob('*'):
                    if file_path.is_file() and file_path.stat().st_size > 0:
                        futures.append(executor.submit(self._hash_file, file_path, file_hashes))
            
            for future in as_completed(futures):
                future.result()
        
        # Agrupar duplicados
        for file_hash, file_paths in file_hashes.items():
            if len(file_paths) > 1:
                # Agrupar por proyecto
                projects = set()
                for path in file_paths:
                    for proj_path in self.projects.keys():
                        if str(path).startswith(proj_path):
                            projects.add(self.projects[proj_path].name)
                            break
                
                if len(projects) > 1:
                    self.duplicates[file_hash] = file_paths
                    
                    # Crear grupo de duplicados
                    group_name = f"dup_{len(self.duplicate_groups)}"
                    self.duplicate_groups[group_name] = list(projects)
                    
                    for proj_name in projects:
                        for proj_path, metadata in self.projects.items():
                            if metadata.name == proj_name:
                                metadata.duplicate_group = group_name
        
        self.report['duplicates_found'] = len(self.duplicates)
        logger.info(f"📊 Duplicados encontrados: {len(self.duplicates)} grupos")
    
    def _hash_file(self, file_path: Path, file_hashes: Dict):
        """Hash de archivo individual"""
        file_hash = self.calculate_hash(file_path)
        if file_hash:
            file_hashes[file_hash].append(file_path)
    
    def consolidate_monorepo(self, target_path: str):
        """Consolidación en monorepo enterprise (Google/Meta style)"""
        logger.info("🏗️  Consolidando en monorepo enterprise...")
        
        target = Path(target_path)
        target.mkdir(parents=True, exist_ok=True)
        
        # Estructura monorepo enterprise
        structure = {
            'apps': {},           # Aplicaciones principales
            'packages': {},       # Librerías compartidas
            'tools': {},          # Herramientas internas
            'services': {},       # Microservicios backend
            'docs': {},           # Documentación
            'scripts': {},        # Scripts de automatización
            'infra': {},          # Infraestructura (Docker, K8s, Terraform)
        }
        
        # Clasificar proyectos
        for project_path, metadata in self.projects.items():
            category = self._classify_for_monorepo(metadata)
            
            if category in structure:
                structure[category][metadata.name] = metadata
        
        # Crear estructura
        for category, projects in structure.items():
            category_path = target / category
            category_path.mkdir(exist_ok=True)
            
            for project_name, metadata in projects.items():
                logger.info(f"📦 Consolidando {project_name} en {category}/")
                
                dest_path = category_path / project_name
                if not dest_path.exists():
                    shutil.copytree(
                        metadata.path, 
                        dest_path, 
                        ignore=shutil.ignore_patterns('node_modules', '.git', '__pycache__', 'venv'),
                        dirs_exist_ok=True
                    )
                    self.report['actions_taken'].append(f"Consolidado: {project_name} -> {category}/")
        
        # Generar configuración enterprise
        self._generate_monorepo_config(target)
        self._generate_workspace_config(target)
        self._generate_turborepo_config(target)
        self._generate_github_actions(target)
        self._generate_docker_config(target)
        self._generate_terraform_config(target)
        
        logger.info(f"✅ Monorepo consolidado en: {target}")
    
    def _classify_for_monorepo(self, metadata: ProjectMetadata) -> str:
        """Clasificación inteligente para monorepo"""
        name = metadata.name.lower()
        
        # Aplicaciones principales
        if any(kw in name for kw in ['web', 'app', 'portal', 'studio', 'experience', 'platform']):
            return 'apps'
        
        # Librerías compartidas
        elif any(kw in name for kw in ['lib', 'core', 'shared', 'utils', 'sdk', 'common']):
            return 'packages'
        
        # Herramientas
        elif any(kw in name for kw in ['tool', 'cli', 'script', 'automation', 'generator']):
            return 'tools'
        
        # Servicios backend
        elif any(kw in name for kw in ['api', 'backend', 'service', 'server', 'microservice']):
            return 'services'
        
        # Documentación
        elif any(kw in name for kw in ['doc', 'guide', 'wiki', 'docs']):
            return 'docs'
        
        # Infraestructura
        elif any(kw in name for kw in ['infra', 'docker', 'k8s', 'terraform', 'deploy']):
            return 'infra'
        
        else:
            return 'apps'  # Por defecto
