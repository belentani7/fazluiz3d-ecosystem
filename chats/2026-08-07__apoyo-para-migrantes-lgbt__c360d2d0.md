[build]
  command = "npm run build"

[[redirects]]
  from = "/*"

  status = 200

[[headers]]
  for = "/*"

```jsx
import { useEffect, useRef } from 'react';
import { gsap } from 'gsap';
import { ScrollTrigger } from 'gsap/ScrollTrigger';

export default function ScrollReveal({ 
  children, 
  className = "",
  delay = 0,
  direction = "up" // up, down, left, right
}) {
  const ref = useRef();
  
  useEffect(() => {
    const element = ref.current;
    if (!element) return;
    
    const directions = {
      up: { y: 60, x: 0 },
      down: { y: -60, x: 0 },
      left: { x: 60, y: 0 },
      right: { x: -60, y: 0 }
    };
    
    const { x, y } = directions[direction];
    
    gsap.fromTo(element, 
      { opacity: 0, x, y },
      {
        opacity: 1,
        x: 0,
        y: 0,
        duration: 1,
        delay,
        ease: "power3.out",
        scrollTrigger: {
          trigger: element,
          start: "top 85%",
          toggleActions: "play none none none"
        }
      }
    );
  }, [delay, direction]);
  
  return <div ref={ref} className={className}>{children}</div>;
}

        top_p: 0.9,
      }),
    });

    if (!response.ok) throw new Error('Groq API failed');
    const data = await response.json();
    const reply = data.choices[0].message.content;

    return NextResponse.json({ reply, agentName: agent.name });

  } catch (error) {
    console.error('Chat Engine Error:', error);
    return NextResponse.json({ 
      reply: "⚠️ Error de conexión con el servidor. Si es una emergencia real, llama al 112. Para asesoría legal urgente: AMIC-UGT 932 915 600." 
    }, { status: 500 });
  }
}
```

#### 2. Frontend: `chat-engine.js` (Lógica de UI y Estado)
*Este script gestiona la memoria local, la sanitización contra inyección XSS y la animación de escritura.*

```javascript
// chat-engine.js
const ChatEngine = {
  history: [],
  currentAdvisor: 'carlos',
  
  init() {
    this.loadHistory();
    this.bindEvents();
  },

  bindEvents() {
    document.getElementById('chatSendBtn')?.addEventListener('click', () => this.send());
    document.getElementById('chatInput')?.addEventListener('keypress', (e) => {
      if (e.key === 'Enter' && !e.shiftKey) {
        e.preventDefault();
        this.send();
      }
    });
  },

  setAdvisor(key, name, welcomeMsg) {
    this.currentAdvisor = key;
    this.history = [{ role: 'assistant', content: welcomeMsg }];
    this.renderChat();
    this.updateUI(name);
  },

  async send() {
    const input = document.getElementById('chatInput');
    const text = input.value.trim();
    if (!text) return;

    // Sanitización básica
    const safeText = text.replace(/</g, "&lt;").replace(/>/g, "&gt;");
    
    this.history.push({ role: 'user', content: safeText });
    input.value = '';
    this.renderChat();
    this.showTyping();

    try {
      const res = await fetch('/api/chat', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ advisor: this.currentAdvisor, messages: this.history })
      });
      
      const data = await res.json();
      this.hideTyping();
      
      // Renderizado con soporte para Markdown básico (negritas)
      const formattedReply = data.reply.replace(/\*\*(.*?)\*\*/g, '<strong>$1</strong>').replace(/\n/g, '<br>');
      this.history.push({ role: 'assistant', content: formattedReply });
      
      // Alerta visual si es crisis
      if (data.isCrisis) {
        this.triggerCrisisUI();
      }

      this.renderChat();
      this.saveHistory();
    } catch (err) {
      this.hideTyping();
      this.history.push({ role: 'assistant', content: '⚠️ Error de red. Contacta recursos reales.' });
      this.renderChat();
    }
  },

  renderChat() {
    const container = document.getElementById('chatContainer');
    if (!container) return;
    
    container.innerHTML = this.history.map(msg => `
      <div class="chat-message ${msg.role}">
        <div class="chat-bubble">${msg.content}</div>
      </div>
    `).join('');
    
    container.scrollTop = container.scrollHeight;
  },

  showTyping() {
    const container = document.getElementById('chatContainer');
    container.innerHTML += `<div id="typing-indicator" class="chat-message bot"><div class="chat-bubble typing-dots">...</div></div>`;
    container.scrollTop = container.scrollHeight;
  },

  hideTyping() {
    document.getElementById('typing-indicator')?.remove();
  },

  triggerCrisisUI() {
    // Cambiar borde del chat a rojo en caso de emergencia detectada
    const container = document.getElementById('chatContainer');
    container.style.borderColor = '#ef4444';
    container.style.boxShadow = '0 0 20px rgba(239, 68, 68, 0.3)';
    setTimeout(() => {
      container.style.borderColor = '';
      container.style.boxShadow = '';
    }, 10000);
  },

  saveHistory() {
    // Guardar solo las últimas 10 interacciones para no saturar localStorage
    const toSave = this.history.slice(-10);
    localStorage.setItem(`chat_history_${this.currentAdvisor}`, JSON.stringify(toSave));
  },

  loadHistory() {
    const saved = localStorage.getItem(`chat_history_${this.currentAdvisor}`);
    if (saved) this.history = JSON.parse(saved);
  },

  updateUI(name) {
    document.getElementById('advisorName').textContent = name;
  }
};

// Inicializar cuando el DOM esté listo
document.addEventListener('DOMContentLoaded', () => ChatEngine.init());
```

### GLOSARIO TÉCNICO Y LEGAL

*   **Edge Function:** Código que se ejecuta en servidores perimetrales (CDN) globales, reduciendo la latencia a milisegundos. Vital para usuarios con conexiones móviles precarias en L'Hospitalet.
*   **CatSalut (TSI):** Sistema de salud catalán. La *Instrucción 10/2018* garantiza la asistencia sanitaria a extranjeros no registrados ni autorizados a residir, siempre que estén empadronados.

*   **RAG (Retrieval-Augmented Generation):** Arquitectura donde la IA consulta una base de datos antes de responder. (En esta versión MVP usamos *In-Context Prompting* por velocidad, pero el sistema está preparado para conectar a una base de datos vectorial como Pinecone en la Fase 2).

### REFLEXIÓN PROSPECTIVA
La intersección entre la inteligencia artificial y la vulnerabilidad migratoria no es un problema de código, es un problema de **diseño de confianza**.
