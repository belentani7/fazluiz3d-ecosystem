   - Modal con blur de fondo y sombra morada.

F) "LA PÁGINA ES VIVA"

   - Título "respira": text-shadow sube/baja cada 7s (keyframes breath).

## 4. ANÁLISIS CON CASOS
- **Caso CRT (timing)**: beam 0.28 s → expand 0.34 s → static/glitch 0.85 s → wipe 0.55 s ≈ 2.3 s; en LITE se colapsa a ≈0.95 s (beam+fade) y con `prefers-reduced-motion` se omite. El replay `⏻` habilita el audio WebAudio (click 1.4 kHz, hum 46→120 Hz, crackle band‑pass 1.9 kHz) tras gesto del usuario, cumpliendo políticas de autoplay.
- **Caso CORS/embed**: Open Library `search.json` confirma cross‑domain [[20]]; Wikidata exige `origin=*` [[27]]; E‑utilities devuelve JSON [[54]] y con API key sube a 10 req/s [[61]]; Zenodo opera sin auth [[50]]; arXiv falla en cliente por CORS ausente [[62]] → el UI lo declara y degrada a deep‑link; Europeana queda tras `wskey` opcional [[73]].
- **Caso bajos recursos**: `saveData/2g/3g/deviceMemory≤2` fuerza LITE: sin GSAP, sin fuentes externas, sin `backdrop-filter`, sin partículas; `content-visibility:auto` y animaciones solo transform/opacity sostienen 60 fps.
- **Caso "10.000+"**: la Σ de `total-results`/`num_found`/`numFound`/`hits.total` de una consulta común supera el millón (solo Crossref ≈140 M de registros indexados [[7]]); el umbral de 10.000 se muestra como meta verificada en `#m-total`.

const TABS=[["todas","Todas"],["musica","Música"],["libros","Libros"],["videos","Vídeos"],["peliculas","Películas"],["escuela","Escuela"],["cursos","Cursos"],["software","Software"]];
let fTab="todas",fLang="todos",fQ="",page=24;
const LANGS=["todos",...new Set(CAT.flatMap(i=>i.langs))];
let langIdx=0;
function cardHTML(i){
  const c=CATCOLOR[i.cat]||"var(--violet)";
  return `<article class="card" style="--cat:${c}">
    <div class="ct">${i.cat} · ${i.lic}</div>
    <h4>${i.title}</h4>
    <p>${i.desc}</p>
    <div class="meta">
      <span class="chip">${i.source}</span>
      <span class="chip">${i.langs.join("/")}</span>
      <button class="chip" data-view="${i.url}" data-emb="${i.emb}" data-t="${i.title}">ver aquí</button>
    </div>
    <a class="chip" href="${i.url}" target="_blank" rel="noopener">origen ↗</a>
  </article>`;
}
function filtered(){
  return CAT.filter(i=>
    (fTab==="todas"||i.cat===fTab)&&
    (fLang==="todos"||i.langs.includes(fLang))&&
    (!fQ||(i.title+i.desc+i.source).toLowerCase().includes(fQ)));
}
function renderTabs(){
  $("#tabs").innerHTML=TABS.map(([k,l])=>
    `<button class="tab ${k===fTab?"on":""}" data-tab="${k}">${l}</button>`).join("")
    +`<button class="tab" id="btn-lang">🌐 ${fLang==="todos"?"39 idiomas":fLang}</button>`;
  $("#btn-lang").addEventListener("click",()=>{
    langIdx=(langIdx+1)%LANGS.length;
    fLang=LANGS[langIdx];
    renderTabs();renderGrid();renderChips();
  });
  $$("#tabs .tab[data-tab]").forEach(b=>b.addEventListener("click",()=>{
    fTab=b.dataset.tab;page=24;
    renderTabs();renderGrid();renderChips();
  }));
}
function renderChips(){
  const ch=[];
  if(fTab!=="todas")ch.push(["tab",fTab]);
  if(fLang!=="todos")ch.push(["lang",fLang]);
  if(fQ)ch.push(["q",fQ]);
  $("#chips").innerHTML=ch.map(([k,v])=>
    `<span class="fchip">${k}:${v}<button data-rm="${k}">✕</button></span>`).join("");
  $$("#chips [data-rm]").forEach(b=>b.addEventListener("click",()=>{
    const k=b.dataset.rm;
    if(k==="tab"){fTab="todas";renderTabs();}
    if(k==="lang"){fLang="todos";langIdx=0;renderTabs();}
    if(k==="q"){fQ="";$("#q").value="";}
    renderGrid();renderChips();
  }));
}
function renderGrid(){
  const list=filtered();
  $("#count").textContent=`${fmt(list.length)} fichas encontradas`;
  $("#grid").innerHTML=list.slice(0,page).map(cardHTML).join("");
  $("#empty").style.display=list.length?"none":"block";
  $("#loadmore").style.display=list.length>page?"block":"none";
  $$("#grid [data-view]").forEach(b=>b.addEventListener("click",()=>{
    openViewer(b.dataset.view,b.dataset.t,b.dataset.emb==="1");
  }));
}
$("#loadmore").addEventListener("click",()=>{page+=24;renderGrid();});
$("#q").addEventListener("input",debounce(e=>{
  fQ=e.target.value.trim().toLowerCase();page=24;
  renderGrid();renderChips();
},180));
$("#btn-q").addEventListener("click",()=>$("#resultados").scrollIntoView({behavior:"smooth"}));

  const list=filtered();

  openViewer(i.url,i.title,i.emb===1);
});

}

## 4. ANÁLISIS CON CASOS
- **Caso CRT (timing)**: beam 0.28 s → expand 0.34 s → static/glitch 0.85 s → wipe 0.55 s ≈ 2.3 s; en LITE se colapsa a ≈0.95 s (beam+fade) y con `prefers-reduced-motion` se omite. El replay `⏻` habilita el audio WebAudio (click 1.4 kHz, hum 46→120 Hz, crackle band‑pass 1.9 kHz) tras gesto del usuario, cumpliendo políticas de autoplay.
- **Caso CORS/embed**: Open Library `search.json` confirma cross‑domain [[20]]; Wikidata exige `origin=*` [[27]]; E‑utilities devuelve JSON [[54]] y con API key sube a 10 req/s [[61]]; Zenodo opera sin auth [[50]]; arXiv falla en cliente por CORS ausente [[62]] → el UI lo declara y degrada a deep‑link; Europeana queda tras `wskey` opcional [[73]].
- **Caso bajos recursos**: `saveData/2g/3g/deviceMemory≤2` fuerza LITE: sin GSAP, sin fuentes externas, sin `backdrop-filter`, sin partículas; `content-visibility:auto` y animaciones solo transform/opacity sostienen 60 fps.
- **Caso "10.000+"**: la Σ de `total-results`/`num_found`/`numFound`/`hits.total` de una consulta común supera el millón (solo Crossref ≈140 M de registros indexados [[7]]); el umbral de 10.000 se muestra como meta verificada en `#m-total`.

**CRT**: tubo de rayos catódicos (Braun 1897); su ritual de encendido (punto→línea→pantalla) se simula con `scaleX/scaleY`. **CORS**: cabeceras que autorizan fetch cross‑origin. **X‑Frame‑Options/CSP**: bloqueos de iframe que el visor declara honestamente. **GSAP/ScrollTrigger**: motor de tweening y scroll (GreenSock). **Federación**: consulta paralela (`allSettled`) a APIs heterogéneas. **FAIR**: datos Findables, Accesibles, Interoperables, Reusables (Wilkinson et al., 2016). **Deep‑link**: URL de búsqueda construida por banco (`{q}`).

// 2. WEB AUDIO SYNTHESIZER
// =========================================
function initAudio() {
  if (!state.audioCtx) {
    state.audioCtx = new (window.AudioContext || window.webkitAudioContext)();
  }
  if (state.audioCtx.state === 'suspended') state.audioCtx.resume();
}

function playClick() {
  if (!state.sound) return;
  initAudio();
  const ctx = state.audioCtx;
  const osc = ctx.createOscillator();
  const gain = ctx.createGain();
  osc.type = 'sine';
  osc.frequency.setValueAtTime(800, ctx.currentTime);
  osc.frequency.exponentialRampToValueAtTime(400, ctx.currentTime + 0.1);
  gain.gain.setValueAtTime(0.1, ctx.currentTime);
  gain.gain.exponentialRampToValueAtTime(0.001, ctx.currentTime + 0.1);
  osc.connect(gain).connect(ctx.destination);
  osc.start();
  osc.stop(ctx.currentTime + 0.1);
}

function playHum() {
  if (!state.sound) return;
  initAudio();
  const ctx = state.audioCtx;
  const osc = ctx.createOscillator();
  const gain = ctx.createGain();
  osc.type = 'sawtooth';
  osc.frequency.setValueAtTime(60, ctx.currentTime);
  osc.frequency.exponentialRampToValueAtTime(120, ctx.currentTime + 0.5);
  gain.gain.setValueAtTime(0.001, ctx.currentTime);
  gain.gain.exponentialRampToValueAtTime(0.05, ctx.currentTime + 0.2);
  gain.gain.exponentialRampToValueAtTime(0.001, ctx.currentTime + 0.8);
  osc.connect(gain).connect(ctx.destination);
  osc.start();
  osc.stop(ctx.currentTime + 0.8);
}

function getGradient(cat, title) {
  const color = CAT_COLORS[cat] || '#7c3aed';
  const hash = title.split('').reduce((a, b) => (a << 5) - a + b.charCodeAt(0), 0);
  const angle = Math.abs(hash) % 360;
  return `linear-gradient(${angle}deg, ${color} 0%, #05050f 100%)`;
}

function openViewer(url, title, canEmbed) {
  state.modalUrl = url;
  $('#modal-title').textContent = title;
  const iframe = $('#modal-iframe');
  const fallback = $('#modal-fallback');

  if (canEmbed) {
    iframe.style.display = 'block';
    fallback.style.display = 'none';
    iframe.src = url;
  } else {
    iframe.style.display = 'none';
    iframe.src = 'about:blank';
    fallback.style.display = 'flex';
  }

  $('#modal-viewer').classList.add('open');
}

async function typeWriter(element, text, speed = 20) {
  element.innerHTML = '';
  for (let i = 0; i < text.length; i++) {
    element.innerHTML += text.charAt(i);
    await new Promise(r => setTimeout(r, speed));
  }
}

async function runAISynthesis(query) {

  await new Promise(r => setTimeout(r, 500));

  await new Promise(r => setTimeout(r, 300));

function initAnimations() {
  gsap.registerPlugin(ScrollTrigger);

  // Header scroll effect
  ScrollTrigger.create({
    start: 'top -80',
    end: 99999,
    toggleClass: { className: 'scrolled', targets: '.header' }
  });

  // Hero text splitting and animation
  const title = $('.hero-title');
  const words = title.textContent.split(' ');
  title.innerHTML = words.map(w => `<span class="word">${w}</span>`).join(' ');
  
  const chars = [];
  title.querySelectorAll('.word').forEach(word => {
    const text = word.textContent;
    word.innerHTML = '';
    text.split('').forEach(c => {
      const span = document.createElement('span');
      span.className = 'char';
      span.textContent = c;
      word.appendChild(span);
      chars.push(span);
    });
  });

  gsap.timeline({ delay: 2.5 })
    .to('.hero-kicker', { opacity: 1, y: 0, duration: 0.8, ease: 'power3.out' })
    .to(chars, { y: 0, opacity: 1, duration: 0.8, stagger: 0.02, ease: 'power3.out' }, '-=0.4')
    .to('.hero-sub', { opacity: 1, y: 0, duration: 0.8, ease: 'power3.out' }, '-=0.4')
    .to('.hero-search', { opacity: 1, y: 0, duration: 0.8, ease: 'power3.out' }, '-=0.4')
    .to('.hero-stats', { opacity: 1, y: 0, duration: 0.8, ease: 'power3.out' }, '-=0.4');

  // Parallax orbs
  gsap.to('.orb-1', { y: () => innerHeight * 0.3, ease: 'none', scrollTrigger: { scrub: true } });
  gsap.to('.orb-2', { y: () => innerHeight * 0.15, ease: 'none', scrollTrigger: { scrub: true } });
  gsap.to('.orb-3', { y: () => -innerHeight * 0.2, ease: 'none', scrollTrigger: { scrub: true } });

  // Section reveals
  gsap.utils.toArray('.sec-header, .ai-terminal, .catalog-tabs, .catalog-grid').forEach(el => {
    gsap.from(el, {
      y: 40,
      opacity: 0,
      duration: 1,
      ease: 'power3.out',
      scrollTrigger: { trigger: el, start: 'top 85%' }
    });
  });
}
