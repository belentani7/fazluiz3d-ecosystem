### 7. src/lib/store.ts
```typescript
import { create } from 'zustand';
import { persist } from 'zustand/middleware';
import type { ExperienceState, DiamondType, SystemMessage } from '@/types';

interface ExperienceStore extends ExperienceState {
  unlockDiamond: (diamond: DiamondType) => void;
  makeChoice: (diamond: DiamondType, choice: string) => void;
  setPhase: (phase: ExperienceState['currentPhase']) => void;
  addMessage: (message: SystemMessage) => void;
  resetExperience: () => void;
  calculateProfile: () => { human: number; artist: number; sinner: number; saint: number; warrior: number };
}

const initialState: ExperienceState = {
  diamonds: {
    human: false,
    artist: false,
    sinner: false,
    saint: false,
    warrior: false,
  },
  choices: {},
  startTime: null,
  completed: false,
  currentPhase: 'home',
  oracleMessages: [
    '> SYSTEM CORE INITIALIZED',
    '> ENTITY: BELENTANI',
    '> AWAITING INTERACTION...',
  ],
};

export const useExperienceStore = create<ExperienceStore>()(
  persist(
    (set, get) => ({
      ...initialState,

      unlockDiamond: (diamond) => {
        set((state) => ({
          diamonds: { ...state.diamonds, [diamond]: true },
          oracleMessages: [
            ...state.oracleMessages,
            `> DIAMOND UNLOCKED: ${diamond.toUpperCase()}`,
            `> FRAGMENT ${Object.values(state.diamonds).filter(Boolean).length + 1} OF 5 RECOVERED`,
          ],
        }));
      },

      makeChoice: (diamond, choice) => {
        set((state) => ({
          choices: {
            ...state.choices,
            [diamond]: [...(state.choices[diamond] || []), choice],
          },
        }));
      },

      setPhase: (phase) => {
        set({ currentPhase: phase });
        if (phase === 'portal' && !get().startTime) {
          set({ startTime: Date.now() });
        }
      },

      addMessage: (message) => {
        set((state) => ({
          oracleMessages: [...state.oracleMessages, message.text],
        }));
      },

      resetExperience: () => {
        set(initialState);
      },

      calculateProfile: () => {
        const state = get();
        const profile = {
          human: 50,
          artist: 50,
          sinner: 50,
          saint: 50,
          warrior: 50,
        };

        // Calculate based on choices
        Object.entries(state.choices).forEach(([diamond, choices]) => {
          const key = diamond as DiamondType;
          if (!choices) return;

          choices.forEach((choice) => {
            // Weight choices based on context
            if (key === 'sinner' && (choice === 'DESIRE' || choice === 'CONFESS')) {
              profile.sinner += 15;
            }
            if (key === 'saint' && choice.includes('forgive')) {
              profile.saint += 15;
            }
            if (key === 'warrior' && choice === 'FIGHT') {
              profile.warrior += 15;
            }
            if (key === 'artist' && choice === 'CREATE') {
              profile.artist += 15;
            }
            if (key === 'human' && choice === 'REMEMBER') {
              profile.human += 15;
            }
          });
        });

        // Normalize to 0-100
        Object.keys(profile).forEach((key) => {
          profile[key as keyof typeof profile] = Math.min(100, Math.max(0, profile[key as keyof typeof profile]));
        });

        return profile;
      },
    }),
    {
      name: 'belentani-judas-experience',
    }
  )
);
```

'use client';

import { useState, useEffect } from 'react';
import { motion, AnimatePresence } from 'framer-motion';
import Portal from '@/components/experience/Portal';
import Constellation from '@/components/experience/Constellation';
import FinalProfile from '@/components/experience/FinalProfile';
import { useExperienceStore } from '@/lib/store';

export default function JudasExperience() {
  const [phase, setPhase] = useState<'intro' | 'portal' | 'constellation' | 'profile'>('intro');
  const { diamonds, setPhase: setStorePhase } = useExperienceStore();

  useEffect(() => {
    const timer = setTimeout(() => {
      setPhase('portal');
      setStorePhase('portal');
    }, 2000);
    return () => clearTimeout(timer);
  }, []);

  const allDiamondsUnlocked = Object.values(diamonds).every(Boolean);

  return (
    <div className="min-h-screen bg-void-black">
      <AnimatePresence mode="wait">
        {phase === 'intro' && (
          <motion.div
            key="intro"
            initial={{ opacity: 0 }}
            animate={{ opacity: 1 }}
            exit={{ opacity: 0 }}
            className="h-screen flex items-center justify-center"
          >
            <div className="text-center">
              <motion.div
                animate={{ scale: [1, 1.2, 1], opacity: [0.5, 1, 0.5] }}
                transition={{ duration: 2, repeat: Infinity }}
                className="w-4 h-4 bg-deep-ruby rounded-full mb-8 mx-auto"
              />
              <p className="text-xl text-gray-400">ENTERING THE EXPERIENCE...</p>
            </div>
          </motion.div>
        )}

        {phase === 'portal' && (
          <motion.div
            key="portal"
            initial={{ opacity: 0 }}
            animate={{ opacity: 1 }}
            exit={{ opacity: 0 }}
            onClick={() => {
              setPhase('constellation');
              setStorePhase('constellation');
            }}
          >
            <Portal />
          </motion.div>
        )}

        {phase === 'constellation' && (
          <motion.div
            key="constellation"
            initial={{ opacity: 0 }}
            animate={{ opacity: 1 }}
            exit={{ opacity: 0 }}
          >
            <Constellation />

                animate={{ opacity: 1, y: 0 }}
                className="fixed bottom-24 left-1/2 transform -translate-x-1/2"
              >
                <button
                  onClick={() => {
                    setPhase('profile');
                    setStorePhase('profile');
                  }}
                  className="px-8 py-3 bg-deep-ruby text-pearl-white font-semibold tracking-wider hover:bg-blood-red transition-colors red-glow"
                >
                  VIEW YOUR PROFILE

        {phase === 'profile' && (

            animate={{ opacity: 1 }}
            exit={{ opacity: 0 }}
          >
            <FinalProfile />
          </motion.div>

      </AnimatePresence>
    </div>
  );
}
