## 🖥️ 2. El CLI (`cli.py`)

El motor de consulta que busca en el índice y devuelve recomendaciones formateadas.

```python
#!/usr/bin/env python3
# cli.py - Meta-Skill Index CLI

import sys
import json
import yaml
import click
from pathlib import Path
from rich.console import Console
from rich.table import Table
from rich.panel import Panel
from rich.markdown import Markdown
from fuzzywuzzy import process
from typing import Dict, List, Optional

console = Console()

# Configuración
CONFIG_DIR = Path.home() / '.meta-skill'
INDEX_PATH = CONFIG_DIR / 'index.yaml'

def load_index() -> Optional[Dict]:
    """Carga el índice desde el archivo YAML."""
    if not INDEX_PATH.exists():
        console.print(f"[red]Error:[/red] Index not found at {INDEX_PATH}")
        console.print("[yellow]Run 'meta-skill init' to create it.[/yellow]")
        return None
    try:
        with open(INDEX_PATH, 'r', encoding='utf-8') as f:
            return yaml.safe_load(f)
    except Exception as e:
        console.print(f"[red]Error loading index:[/red] {e}")
        return None

def format_recommendation(task: Dict, user_prompt: str = "") -> str:
    """Formatea una recomendación como texto legible."""
    rec = task.get('recommendations', {})
    rec_text = f"""
## 📋 Tarea: {task.get('description', 'Sin descripción')}

### 🎯 Recomendación:
- **CLI**: `{rec.get('cli', 'N/A')}`
- **API**: `{rec.get('api', 'N/A')}`
- **Modelo**: `{rec.get('model', 'N/A')}`
- **Skill**: `{rec.get('skill', 'N/A')}`
- **Lenguaje**: `{rec.get('language', 'N/A')}`
- **Nivel de razonamiento**: `{rec.get('reasoning_level', 'N/A')}`
- **Función**: `{rec.get('function', 'N/A')}`
- **Herramientas**: `{', '.join(rec.get('tools', []))}`

### ✏️ Prompt de ejemplo:
```
{rec.get('example_prompt', 'No disponible')}
```

### 🚀 Para ejecutar:
```bash
{rec.get('cli', '')} "{rec.get('example_prompt', '')}"
```
"""
    if user_prompt:
        rec_text += f"\n### 📝 Prompt personalizado:\n```\n{user_prompt}\n```\n"
    return rec_text

@click.group()
def cli():
    """Meta-Skill Index - Encuentra la mejor combinación de herramientas para tu tarea."""
    pass

@cli.command()
def init():
    """Inicializa el índice en ~/.meta-skill/index.yaml"""
    CONFIG_DIR.mkdir(exist_ok=True)
    # Copiar índice por defecto (si no existe)
    if not INDEX_PATH.exists():
        # Creamos un índice básico (el contenido se genera después)
        default_index = {
            "version": "1.0.0",
            "last_updated": "2026-07-18",
            "areas": ["frontend", "backend", "security", "data", "devops", "ai_ml", "testing", "documentation"],
            "tasks": [],
            "fallback": {
                "cli": "mimo",
                "api": "openrouter",
                "model": "claude-3-sonnet",
                "skill": "general-assistant",
                "language": "any",
                "reasoning_level": "medium",
                "function": "general_task",
                "tools": [],
                "example_prompt": "Ayuda a resolver esta tarea: {user_prompt}"
            }
        }
        with open(INDEX_PATH, 'w', encoding='utf-8') as f:
            yaml.dump(default_index, f, allow_unicode=True, default_flow_style=False)
        console.print(f"[green]✓[/green] Index initialized at {INDEX_PATH}")
        console.print("[yellow]Add your own tasks by editing index.yaml[/yellow]")
    else:
        console.print(f"[yellow]Index already exists at {INDEX_PATH}[/yellow]")

@cli.command()
@click.argument('query', required=False)
@click.option('--format', '-f', type=click.Choice(['text', 'json', 'markdown']), default='text')
@click.option('--execute', '-e', is_flag=True, help='Execute the recommended CLI command')
def search(query: str, format: str, execute: bool):
    """Busca tareas que coincidan con la consulta."""
    index = load_index()
    if not index:
        return

    tasks = index.get('tasks', [])
    fallback = index.get('fallback', {})

    if not query:
        # Mostrar todas las áreas y tareas disponibles
        table = Table(title="📚 Áreas disponibles")
        table.add_column("Área", style="cyan")
        table.add_column("Tareas", style="green")
        areas = {}
        for task in tasks:
            area = task.get('area', 'other')
            if area not in areas:
                areas[area] = []
            areas[area].append(task.get('description', 'Sin descripción')[:50])
        for area, task_list in areas.items():
            table.add_row(area, f"{len(task_list)} tareas")
        console.print(table)
        console.print("\n[bold]Usa:[/bold] meta-skill search \"tu descripción de tarea\"")
        return

    # Buscar coincidencias
    descriptions = [t.get('description', '') for t in tasks]
    matches = process.extract(query, descriptions, limit=5, scorer=fuzzywuzzy.fuzz.token_sort_ratio)

    if not matches or matches[0][1] < 30:
        console.print("[yellow]No se encontraron coincidencias exactas. Usando fallback.[/yellow]")
        task = {
            "description": query,
            "recommendations": fallback
        }
        if format == 'json':
            console.print(json.dumps(task, indent=2, ensure_ascii=False))
        else:
            console.print(format_recommendation(task, query))
        return

    # Mostrar los resultados
    results = []
    for desc, score in matches[:3]:  # Top 3
        task = next(t for t in tasks if t.get('description') == desc)
        results.append((task, score))

    if format == 'json':
        console.print(json.dumps([{'task': t[0], 'score': t[1]} for t in results], indent=2, ensure_ascii=False))
        return

    console.print(f"\n[bold cyan]🔍 Encontradas {len(results)} coincidencias[/bold cyan]\n")
    for i, (task, score) in enumerate(results, 1):
        console.print(Panel(
            format_recommendation(task),
            title=f"[bold]Opción {i} (Score: {score}%)[/bold]",
            border_style="blue"
        ))

    # Preguntar al usuario
    if len(results) > 1:
        choice = click.prompt("Selecciona una opción (1-3)", type=int, default=1)
        if 1 <= choice <= len(results):
            task = results[choice-1][0]
            rec = task.get('recommendations', {})
            if execute:
                cmd = rec.get('cli', '')
                if cmd:
                    console.print(f"[bold green]▶ Ejecutando:[/bold green] {cmd} \"{query}\"")
                    # Usar subprocess para ejecutar
                    import subprocess
                    try:
                        subprocess.run([cmd, query], check=True)
                    except Exception as e:
                        console.print(f"[red]Error:[/red] {e}")
                else:
                    console.print("[red]No se encontró CLI para ejecutar.[/red]")
            else:
                console.print("\n[bold green]✅ Recomendación seleccionada:[/bold green]")
                console.print(format_recommendation(task, query))
        else:
            console.print("[red]Opción inválida.[/red]")

@cli.command()
@click.argument('task_id')
def show(task_id):
    """Muestra una tarea específica por su ID."""
    index = load_index()
    if not index:
        return
    tasks = index.get('tasks', [])
    task = next((t for t in tasks if t.get('id') == task_id), None)
    if task:
        console.print(format_recommendation(task))
    else:
        console.print(f"[red]Tarea '{task_id}' no encontrada.[/red]")

@cli.command()
@click.argument('area')
def list_area(area):
    """Lista todas las tareas de un área."""
    index = load_index()
    if not index:
        return
    tasks = [t for t in index.get('tasks', []) if t.get('area') == area]
    if not tasks:
        console.print(f"[yellow]No se encontraron tareas en el área '{area}'[/yellow]")
        return
    table = Table(title=f"Tareas en {area}")
    table.add_column("ID", style="cyan")
    table.add_column("Descripción", style="green")
    table.add_column("CLI", style="yellow")
    table.add_column("Modelo", style="magenta")
    for task in tasks:
        rec = task.get('recommendations', {})
        table.add_row(
            task.get('id', ''),
            task.get('description', '')[:50] + '...',
            rec.get('cli', ''),
            rec.get('model', '')
        )
    console.print(table)

if __name__ == '__main__':
    cli()
```

# Meta-Skill Index

Este es el **skill de todos los skills**. Antes de empezar cualquier tarea, consulta este skill para saber qué herramientas usar.

## ¿Para qué sirve?

- Te recomienda el CLI, API, modelo, skill, lenguaje y nivel de razonamiento óptimo para tu tarea.
- Te da un prompt de ejemplo listo para usar.
- Te ahorra tiempo decidiendo qué herramienta usar entre todas las que tienes instaladas.

### Desde la línea de comandos (CLI)

```bash
# Consultar el índice
meta-skill search "necesito hacer animaciones GSAP"

meta-skill search

meta-skill list-area frontend

```bash

#!/bin/bash
# install.sh - Instalador del Meta-Skill Index

set -e

echo "🚀 Instalando Meta-Skill Index..."

# Crear directorio de configuración
mkdir -p ~/.meta-skill

# Copiar el índice
if [ -f "index.yaml" ]; then
    cp index.yaml ~/.meta-skill/index.yaml
    echo "✅ Índice copiado a ~/.meta-skill/index.yaml"
else
    echo "❌ index.yaml no encontrado. Ejecuta 'meta-skill init' para crearlo."
fi

# Instalar dependencias Python
echo "📦 Instalando dependencias Python..."
pip install click pyyaml rich fuzzywuzzy python-Levenshtein

### 3. Capas de la Herramienta

Para que sea usable y extensible, la herramienta debe tener tres capas:

#### A. Capa de Datos (El Índice)
- Un archivo YAML/JSON con la estructura anterior.
- Puede ser editado manualmente o generado con scripts que scrapeen documentación oficial.
- **Ventaja**: Es texto plano, fácil de versionar y colaborar.

#### B. Capa de Consulta (El Motor)
- Un CLI (puede ser Python o Node.js) que:
    - Recibe una descripción de tarea (ej. "necesito hacer animaciones GSAP").
    - Busca en el índice la tarea más similar (usando un embedding simple o keywords).
    - Devuelve la recomendación completa.
- **Ventaja**: No requiere IA; puede funcionar con búsqueda por palabras clave.

#### C. Capa de Integración (El Skill)
- Un skill para agentes (Mimo, Claude, Hermes) que:
    - Recibe la tarea del usuario.
    - Consulta el motor de recomendación.
    - Devuelve la sugerencia formateada y, opcionalmente, genera el prompt exacto.
- **Ventaja**: Se integra en el flujo de trabajo de los asistentes existentes.

```python
# cli.py
import click
import yaml
from pathlib import Path
from rich.console import Console
from rich.table import Table
from fuzzywuzzy import process

console = Console()
INDEX_PATH = Path.home() / '.project-index' / 'index.yaml'

def load_index():
    if not INDEX_PATH.exists():
        console.print("[red]Index not found. Run 'project-index init' first.[/red]")
        return None
    with open(INDEX_PATH) as f:
        return yaml.safe_load(f)

@click.group()
def cli():
    """Project Index - encuentra la mejor combinación de herramientas para tu tarea."""
    pass

@cli.command()
def init():
    """Inicializa el índice en ~/.project-index/"""
    INDEX_PATH.parent.mkdir(exist_ok=True)
    # Copia el índice por defecto
    console.print("[green]Index initialized at ~/.project-index/index.yaml[/green]")

@cli.command()
@click.argument('query')
def search(query):
    """Busca tareas que coincidan con la consulta."""
    index = load_index()
    if not index:
        return

    tasks = index.get('tasks', [])
    # Buscar coincidencias por descripción
    matches = process.extract(query, [t['description'] for t in tasks], limit=5)

    table = Table(title="Recomendaciones encontradas")
    table.add_column("ID", style="cyan")
    table.add_column("Área", style="green")
    table.add_column("Descripción")
    table.add_column("CLI", style="yellow")
    table.add_column("Modelo", style="magenta")

    for match in matches:
        desc = match[0]
        score = match[1]
        task = next(t for t in tasks if t['description'] == desc)
        rec = task['recommendations']
        table.add_row(
            task['id'],
            task['area'],
            desc[:50] + "...",
            rec.get('cli', ''),
            rec.get('model', '')
        )

    console.print(table)

if __name__ == '__main__':
    cli()
```

### 🧠 1. El Motor de Decisión: ¿Qué Afecta la Ejecución?

La adaptación de la herramienta no es una elección única, sino un equilibrio entre varias variables críticas. La base de datos debe modelar estas dimensiones para cada tarea:

*   **Rendimiento (Quality)**: La capacidad del modelo para resolver la tarea correctamente. Se mide con benchmarks como **SWE-bench** para codificación, **GPQA** para razonamiento, y benchmarks especializados como **DeepSWE** para tareas de ingeniería de software a largo plazo.
*   **Coste (Cost)**: El coste por millón de tokens (entrada/salida). Modelos como `GPT-5.6 Terra` ofrecen reducciones de coste del 60-80% en comparación con modelos más potentes.
*   **Velocidad (Speed/Latencia)**: El tiempo de respuesta, crucial para tareas interactivas o de desarrollo rápido.
*   **Contexto (Context Window)**: La cantidad de información que el modelo puede procesar de una vez. Crítico para proyectos grandes o análisis de código extenso.
*   **Seguridad (Security)**: La capacidad del modelo para generar código seguro y resistente a vulnerabilidades.
*   **Capacidad Agéntica (Agentic Capability)**: La habilidad para usar herramientas, planificar y ejecutar tareas complejas de forma autónoma.

Cada tarea en tu índice ponderará estas variables de forma diferente. Por ejemplo:

*   **Frontend/Animaciones (GSAP)**: Ponderación alta en **Rendimiento Creativo** y **Velocidad**, ponderación media en **Coste**.
*   **Backend/API REST**: Ponderación alta en **Rendimiento de Codificación** y **Seguridad**, ponderación media en **Coste**.
*   **Seguridad/Auditoría de Código**: Ponderación muy alta en **Seguridad** y **Rendimiento de Razonamiento**, ponderación baja en **Velocidad**.
*   **Tareas Simples/Generación de Documentación**: Ponderación alta en **Velocidad** y **Coste**, ponderación media en **Rendimiento**.

#### b) Mecanismo de Actualización Automática

Para que la base de datos se autocorrija, necesitas un script que:

1.  **Scrapee líderes de benchmarks** públicos como [llm-stats.com](https://llm-stats.com), [LiveBench](https://livebench.ai), o [Klu.ai](https://klu.ai).
2.  **Extraiga los datos** de rendimiento de cada modelo.
3.  **Actualice la base de datos** (`performance_db.yaml`) con los nuevos valores.
4.  **Recalcule las recomendaciones** para cada tarea en `index.yaml` basándose en los nuevos datos.

Este script podría ejecutarse diariamente o semanalmente (vía GitHub Actions), asegurando que tu índice esté siempre actualizado.

### 🔄 3. El Mecanismo de Autocorrección (El "Cerebro" Adaptativo)

La base de datos no solo almacena datos, sino que también **aprende de las interacciones de los usuarios**. Cada vez que un usuario usa `meta-skill search` y selecciona una opción, o ejecuta `--execute` con un modelo, el sistema registra:

*   **Éxito/Fracaso**: ¿El modelo resolvió la tarea correctamente? (Esto podría ser feedback implícito, como "el usuario no volvió a preguntar", o explícito, como una calificación).
*   **Tiempo de respuesta**: ¿Fue rápido?
*   **Coste real**: ¿Cuánto costó la ejecución?

Estos datos alimentan un **bucle de retroalimentación** que ajusta las ponderaciones de las variables para cada tarea. Por ejemplo, si los usuarios eligen constantemente `GPT-5.6 Sol` para tareas de seguridad y obtienen buenos resultados, el sistema aumentará la ponderación de `security_score` para esas tareas.

Este mecanismo se basa en técnicas de **enrutamiento dinámico** y **aprendizaje por refuerzo**, donde el sistema aprende a elegir el mejor modelo para cada situación sin necesidad de intervención manual constante.

### 🧩 4. La Matriz de Decisión Final (Cómo se Unifica Todo)

Para cada tarea en tu `index.yaml`, el sistema realiza los siguientes pasos:

1.  **Recupera la tarea** y sus ponderaciones de variables.
2.  **Consulta la base de datos de rendimiento** para obtener las puntuaciones de todos los modelos disponibles.
3.  **Calcula una puntuación compuesta** para cada modelo, basada en las ponderaciones de la tarea y las puntuaciones del modelo.
4.  **Selecciona el modelo con la puntuación más alta** y devuelve la recomendación completa (CLI, API, skill, etc.).
5.  **Registra la interacción** para futuras actualizaciones.

Este proceso convierte tu `Meta-Skill Index` en un sistema de recomendación inteligente y adaptativo, muy superior a un simple catálogo estático.

def load_index():
    if not INDEX_PATH.exists():

def calculate_best_cli(task):
    """Motor Adaptativo: Calcula el mejor CLI basado en los pesos de la tarea."""
    weights = task.get('weights', {})
    recs = task.get('recommendations', {})
    candidates = [recs.get('primary_cli'), recs.get('fallback_cli')]
    
    best_score = -1
    best_cli = recs.get('primary_cli')
    scores = {}
    
    for cli in candidates:
        if not cli or cli not in MODEL_PERFORMANCE:
            continue
        perf = MODEL_PERFORMANCE[cli]
        # Producto escalar entre los pesos de la tarea y el rendimiento del modelo
        score = sum(weights.get(k, 0) * perf.get(k, 50) for k in weights)
        scores[cli] = score
        if score > best_score:
            best_score = score
            best_cli = cli
            
    return best_cli, scores

2. **El Actualizador de Base de Datos (Scraping Gratuito):** Un script que se ejecuta en segundo plano (o bajo demanda) y extrae los datos de fuentes públicas y gratuitas (como la API pública de OpenRouter o tablas de GitHub) para actualizar tu YAML local.

### 🔄 1. El Actualizador Automático (`updater.py`)

Este script consume la **API Pública de OpenRouter** (que es 100% gratuita, no requiere API Key y se actualiza diariamente con los nuevos modelos, sus precios y ventanas de contexto) y cruza esos datos con heurísticas locales para actualizar tu base de datos.

```python
#!/usr/bin/env python3
# updater.py - Actualizador Zero-Token de la Base de Datos de Rendimiento

import yaml
import requests
import re
from pathlib import Path
from rich.console import Console
from rich.progress import Progress

console = Console()
CONFIG_DIR = Path.home() / '.meta-skill'
PERF_DB_PATH = CONFIG_DIR / 'performance_db.yaml'

# API Pública y Gratuita (No requiere API Key)
OPENROUTER_API = "https://openrouter.ai/api/v1/models"

def fetch_public_models():
    """Obtiene la lista de modelos, precios y contextos desde OpenRouter (Gratis)."""
    try:
        response = requests.get(OPENROUTER_API, timeout=10)
        response.raise_for_status()
        return response.json().get('data', [])
    except Exception as e:
        console.print(f"[red]❌ Error al conectar con la fuente pública: {e}[/red]")
        return []

def heuristic_scoring(model_id, context_length, pricing):
    """
    Asigna puntuaciones de rendimiento basadas en el nombre del modelo y metadatos.
    Esto evita usar una API de IA para evaluar a otra IA.
    """
    scores = {
        "reasoning": 50, "coding": 50, "speed": 50, 
        "cost": 50, "context": 50, "agentic": 50, "security": 50
    }
    
    name = model_id.lower()
    
    # Heurística de Contexto
    if context_length >= 1000000: scores["context"] = 100
    elif context_length >= 200000: scores["context"] = 90
    elif context_length >= 128000: scores["context"] = 80
    else: scores["context"] = 60

    # Heurística de Coste (Más barato = Mayor puntuación en 'cost')
    prompt_price = float(pricing.get("prompt", "0"))
    if prompt_price == 0: scores["cost"] = 100
    elif prompt_price < 1.0: scores["cost"] = 90
    elif prompt_price < 5.0: scores["cost"] = 70
    else: scores["cost"] = 40

    # Heurística de Capacidades (Basada en nomenclatura estándar de la industria)
    if "opus" in name or "o1" in name or "gemini-2.5-pro" in name:
        scores["reasoning"] = 95; scores["security"] = 90
    if "sonnet" in name or "gpt-4o" in name:
        scores["reasoning"] = 85; scores["coding"] = 85
    if "coder" in name or "qwen" in name or "deepseek" in name:
        scores["coding"] = 95; scores["agentic"] = 85
    if "flash" in name or "haiku" in name or "mini" in name:
        scores["speed"] = 95; scores["cost"] = 95
        
    return scores

def update_database():
    """Descarga datos reales y actualiza el YAML local."""
    console.print("[bold cyan]🔄 Iniciando actualización Zero-Token...[/bold cyan]")
    
    models_data = fetch_public_models()
    if not models_data:
        return

    local_db = {"models": []}
    
    # Filtramos solo los modelos más relevantes para no saturar la DB local
    top_providers = ["openai", "anthropic", "google", "meta-llama", "qwen", "deepseek"]
    
    with Progress() as progress:
        task = progress.add_task("[green]Procesando modelos...", total=len(models_data))
        
        for model in models_data:
            progress.update(task, advance=1)
            model_id = model.get("id", "")
            
            # Solo procesamos modelos de proveedores principales o gratuitos
            if not any(provider in model_id for provider in top_providers) and ":free" not in model_id:
                continue
                
            context = model.get("context_length", 4096)
            pricing = model.get("pricing", {})
            
            scores = heuristic_scoring(model_id, context, pricing)
            
            local_db["models"].append({
                "id": model_id,
                "context_window": context,
                "prompt_price": pricing.get("prompt", "0"),
                "scores": scores
            })

    # Guardar en el YAML local
    CONFIG_DIR.mkdir(exist_ok=True)
    with open(PERF_DB_PATH, 'w', encoding='utf-8') as f:
        yaml.dump(local_db, f, allow_unicode=True, default_flow_style=False)
        
    console.print(f"[bold green]✅ Base de datos actualizada con {len(local_db['models'])} modelos reales.[/bold green]")
    console.print(f"[yellow]Guardado en: {PERF_DB_PATH}[/yellow]")

if __name__ == "__main__":
    update_database()
```

### ⚙️ 2. Integración en el CLI (`cli.py`)

```python
# ... (Imports anteriores) ...
from updater import update_database, PERF_DB_PATH

# ... (Código anterior) ...

@cli.command()
def update():
    """🔄 Actualiza la base de datos de rendimiento con precios y contextos reales (Zero-Token)."""
    update_database()

def load_performance_db():
    """Carga la base de datos de rendimiento real."""
    if not PERF_DB_PATH.exists():
        console.print("[yellow]⚠️ DB de rendimiento no encontrada. Ejecutando 'meta-skill update'...[/yellow]")
        update_database()
    
    with open(PERF_DB_PATH, 'r', encoding='utf-8') as f:
        data = yaml.safe_load(f)
        
    # Convertimos la lista a un diccionario para búsqueda rápida O(1)
    return {m['id']: m for m in data.get('models', [])}

def calculate_best_cli(task):
    """Motor Adaptativo: Usa la DB real de precios y contextos."""
    weights = task.get('weights', {})
    recs = task.get('recommendations', {})
    candidates = [recs.get('primary_cli'), recs.get('fallback_cli')]
    
    perf_db = load_performance_db()
    best_score = -1
    best_cli = recs.get('primary_cli')
    
    for cli in candidates:
        if not cli: continue
        
        # Buscamos el modelo en nuestra DB real (mapeando nombres de CLI a IDs de modelo)
        # Ejemplo: "zcode" -> "zhipu/glm-4", "claude-code" -> "anthropic/claude-sonnet-4"
        model_id = map_cli_to_model_id(cli) 
        model_data = perf_db.get(model_id, {})
        scores = model_data.get('scores', {})
        
        # Producto escalar
        score = sum(weights.get(k, 0) * scores.get(k, 50) for k in weights)
        
        if score > best_score:
            best_score = score
            best_cli = cli
            
    return best_cli, best_score

def map_cli_to_model_id(cli_name):
    """Mapea el nombre de tu CLI local al ID real de la API pública."""
    mapping = {
        "zcode": "zhipu/glm-4-9b", # Ejemplo
        "claude-code": "anthropic/claude-sonnet-4",
        "qwen": "qwen/qwen-2.5-coder-32b-instruct",
        "autoclaw": "openai/gpt-4o-mini", # AutoClaw suele usar modelos rápidos
        "cursor": "anthropic/claude-3.5-sonnet",

## 🖥️ 2. El CLI (`cli.py`)

El motor de consulta que busca en el índice y devuelve recomendaciones formateadas.

```python
#!/usr/bin/env python3
# cli.py - Meta-Skill Index CLI

import sys
import json
import yaml
import click
from pathlib import Path
from rich.console import Console
from rich.table import Table
from rich.panel import Panel
from rich.markdown import Markdown
from fuzzywuzzy import process
from typing import Dict, List, Optional

console = Console()

# Configuración
CONFIG_DIR = Path.home() / '.meta-skill'
INDEX_PATH = CONFIG_DIR / 'index.yaml'

def load_index() -> Optional[Dict]:
    """Carga el índice desde el archivo YAML."""
    if not INDEX_PATH.exists():
        console.print(f"[red]Error:[/red] Index not found at {INDEX_PATH}")
        console.print("[yellow]Run 'meta-skill init' to create it.[/yellow]")
        return None
    try:
        with open(INDEX_PATH, 'r', encoding='utf-8') as f:
            return yaml.safe_load(f)
    except Exception as e:
        console.print(f"[red]Error loading index:[/red] {e}")
        return None

def format_recommendation(task: Dict, user_prompt: str = "") -> str:
    """Formatea una recomendación como texto legible."""
    rec = task.get('recommendations', {})
    rec_text = f"""
## 📋 Tarea: {task.get('description', 'Sin descripción')}

### 🎯 Recomendación:
- **CLI**: `{rec.get('cli', 'N/A')}`
- **API**: `{rec.get('api', 'N/A')}`
- **Modelo**: `{rec.get('model', 'N/A')}`
- **Skill**: `{rec.get('skill', 'N/A')}`
- **Lenguaje**: `{rec.get('language', 'N/A')}`
- **Nivel de razonamiento**: `{rec.get('reasoning_level', 'N/A')}`
- **Función**: `{rec.get('function', 'N/A')}`
- **Herramientas**: `{', '.join(rec.get('tools', []))}`

### ✏️ Prompt de ejemplo:
```
{rec.get('example_prompt', 'No disponible')}
```

### 🚀 Para ejecutar:
```bash
{rec.get('cli', '')} "{rec.get('example_prompt', '')}"
```
"""
    if user_prompt:
        rec_text += f"\n### 📝 Prompt personalizado:\n```\n{user_prompt}\n```\n"
    return rec_text

@click.group()
def cli():
    """Meta-Skill Index - Encuentra la mejor combinación de herramientas para tu tarea."""
    pass

@cli.command()
def init():
    """Inicializa el índice en ~/.meta-skill/index.yaml"""
    CONFIG_DIR.mkdir(exist_ok=True)
    # Copiar índice por defecto (si no existe)
    if not INDEX_PATH.exists():
        # Creamos un índice básico (el contenido se genera después)
        default_index = {
            "version": "1.0.0",
            "last_updated": "2026-07-18",
            "areas": ["frontend", "backend", "security", "data", "devops", "ai_ml", "testing", "documentation"],
            "tasks": [],
            "fallback": {
                "cli": "mimo",
                "api": "openrouter",
                "model": "claude-3-sonnet",
                "skill": "general-assistant",
                "language": "any",
                "reasoning_level": "medium",
                "function": "general_task",
                "tools": [],
                "example_prompt": "Ayuda a resolver esta tarea: {user_prompt}"
            }
        }
        with open(INDEX_PATH, 'w', encoding='utf-8') as f:
            yaml.dump(default_index, f, allow_unicode=True, default_flow_style=False)
        console.print(f"[green]✓[/green] Index initialized at {INDEX_PATH}")
        console.print("[yellow]Add your own tasks by editing index.yaml[/yellow]")
    else:
        console.print(f"[yellow]Index already exists at {INDEX_PATH}[/yellow]")

@cli.command()
@click.argument('query', required=False)
@click.option('--format', '-f', type=click.Choice(['text', 'json', 'markdown']), default='text')
@click.option('--execute', '-e', is_flag=True, help='Execute the recommended CLI command')
def search(query: str, format: str, execute: bool):
    """Busca tareas que coincidan con la consulta."""
    index = load_index()
    if not index:
        return

    tasks = index.get('tasks', [])
    fallback = index.get('fallback', {})

    if not query:
        # Mostrar todas las áreas y tareas disponibles
        table = Table(title="📚 Áreas disponibles")
        table.add_column("Área", style="cyan")
        table.add_column("Tareas", style="green")
        areas = {}
        for task in tasks:
            area = task.get('area', 'other')
            if area not in areas:
                areas[area] = []
            areas[area].append(task.get('description', 'Sin descripción')[:50])
        for area, task_list in areas.items():
            table.add_row(area, f"{len(task_list)} tareas")
        console.print(table)
        console.print("\n[bold]Usa:[/bold] meta-skill search \"tu descripción de tarea\"")
        return

    # Buscar coincidencias
    descriptions = [t.get('description', '') for t in tasks]
    matches = process.extract(query, descriptions, limit=5, scorer=fuzzywuzzy.fuzz.token_sort_ratio)

    if not matches or matches[0][1] < 30:
        console.print("[yellow]No se encontraron coincidencias exactas. Usando fallback.[/yellow]")
        task = {
            "description": query,
            "recommendations": fallback
        }
        if format == 'json':
            console.print(json.dumps(task, indent=2, ensure_ascii=False))
        else:
            console.print(format_recommendation(task, query))
        return

    # Mostrar los resultados
    results = []
    for desc, score in matches[:3]:  # Top 3
        task = next(t for t in tasks if t.get('description') == desc)
        results.append((task, score))

    if format == 'json':
        console.print(json.dumps([{'task': t[0], 'score': t[1]} for t in results], indent=2, ensure_ascii=False))
        return

    console.print(f"\n[bold cyan]🔍 Encontradas {len(results)} coincidencias[/bold cyan]\n")
    for i, (task, score) in enumerate(results, 1):
        console.print(Panel(
            format_recommendation(task),
            title=f"[bold]Opción {i} (Score: {score}%)[/bold]",
            border_style="blue"
        ))

    # Preguntar al usuario
    if len(results) > 1:
        choice = click.prompt("Selecciona una opción (1-3)", type=int, default=1)
        if 1 <= choice <= len(results):
            task = results[choice-1][0]
            rec = task.get('recommendations', {})
            if execute:
                cmd = rec.get('cli', '')
                if cmd:
                    console.print(f"[bold green]▶ Ejecutando:[/bold green] {cmd} \"{query}\"")
                    # Usar subprocess para ejecutar
                    import subprocess
                    try:
                        subprocess.run([cmd, query], check=True)
                    except Exception as e:
                        console.print(f"[red]Error:[/red] {e}")
                else:
                    console.print("[red]No se encontró CLI para ejecutar.[/red]")
            else:
                console.print("\n[bold green]✅ Recomendación seleccionada:[/bold green]")
                console.print(format_recommendation(task, query))
        else:
            console.print("[red]Opción inválida.[/red]")

@cli.command()
@click.argument('task_id')
def show(task_id):
    """Muestra una tarea específica por su ID."""
    index = load_index()
    if not index:
        return
    tasks = index.get('tasks', [])
    task = next((t for t in tasks if t.get('id') == task_id), None)
    if task:
        console.print(format_recommendation(task))
    else:
        console.print(f"[red]Tarea '{task_id}' no encontrada.[/red]")

@cli.command()
@click.argument('area')
def list_area(area):
    """Lista todas las tareas de un área."""
    index = load_index()
    if not index:
        return
    tasks = [t for t in index.get('tasks', []) if t.get('area') == area]
    if not tasks:
        console.print(f"[yellow]No se encontraron tareas en el área '{area}'[/yellow]")
        return
    table = Table(title=f"Tareas en {area}")
    table.add_column("ID", style="cyan")
    table.add_column("Descripción", style="green")
    table.add_column("CLI", style="yellow")
    table.add_column("Modelo", style="magenta")
    for task in tasks:
        rec = task.get('recommendations', {})
        table.add_row(
            task.get('id', ''),
            task.get('description', '')[:50] + '...',
            rec.get('cli', ''),
            rec.get('model', '')
        )
    console.print(table)

if __name__ == '__main__':
    cli()
```

## 🧠 3. El Skill para Asistentes (`skill/SKILL.md`)

Este archivo define el skill para Mimo, Claude, etc. Lo colocas en `~/.mimo/skills/meta-skill/` o `~/.claude/skills/meta-skill/`.

```markdown
---
name: meta-skill
description: El skill universal que te dice qué CLI, API, modelo, skill y lenguaje usar para cualquier tarea. Va por delante de todos los demás skills.
version: 1.0.0
author: User
tags: [universal, index, recommendation, orchestrator]
---

# Meta-Skill Index

Este es el **skill de todos los skills**. Antes de empezar cualquier tarea, consulta este skill para saber qué herramientas usar.

## ¿Para qué sirve?

- Te recomienda el CLI, API, modelo, skill, lenguaje y nivel de razonamiento óptimo para tu tarea.
- Te da un prompt de ejemplo listo para usar.
- Te ahorra tiempo decidiendo qué herramienta usar entre todas las que tienes instaladas.

## Cómo usarlo

### Desde la línea de comandos (CLI)

```bash
# Consultar el índice
meta-skill search "necesito hacer animaciones GSAP"

# Ver todas las áreas disponibles
meta-skill search

# Ver una tarea específica
meta-skill show frontend-animations-gsap

# Listar tareas de un área
meta-skill list-area frontend
```

### Desde el asistente (Mimo, Claude, etc.)

Usa el comando slash:

```bash
/meta-skill "necesito auditar código en busca de vulnerabilidades"
```

O simplemente describe tu tarea y el skill se activará automáticamente (si está configurado como skill principal).

    cp index.yaml ~/.meta-skill/index.yaml
    echo "✅ Índice copiado a ~/.meta-skill/index.yaml"
else
    echo "❌ index.yaml no encontrado. Ejecuta 'meta-skill init' para crearlo."
fi

### 3. Capas de la Herramienta

Para que sea usable y extensible, la herramienta debe tener tres capas:

#### A. Capa de Datos (El Índice)
- Un archivo YAML/JSON con la estructura anterior.
- Puede ser editado manualmente o generado con scripts que scrapeen documentación oficial.
- **Ventaja**: Es texto plano, fácil de versionar y colaborar.

#### B. Capa de Consulta (El Motor)
- Un CLI (puede ser Python o Node.js) que:
    - Recibe una descripción de tarea (ej. "necesito hacer animaciones GSAP").
    - Busca en el índice la tarea más similar (usando un embedding simple o keywords).
    - Devuelve la recomendación completa.
- **Ventaja**: No requiere IA; puede funcionar con búsqueda por palabras clave.

#### C. Capa de Integración (El Skill)
- Un skill para agentes (Mimo, Claude, Hermes) que:
    - Recibe la tarea del usuario.
    - Consulta el motor de recomendación.
    - Devuelve la sugerencia formateada y, opcionalmente, genera el prompt exacto.
- **Ventaja**: Se integra en el flujo de trabajo de los asistentes existentes.

```python
# cli.py
import click
import yaml
from pathlib import Path
from rich.console import Console
from rich.table import Table
from fuzzywuzzy import process

console = Console()
INDEX_PATH = Path.home() / '.project-index' / 'index.yaml'

def load_index():
    if not INDEX_PATH.exists():
        console.print("[red]Index not found. Run 'project-index init' first.[/red]")
        return None
    with open(INDEX_PATH) as f:
        return yaml.safe_load(f)

@click.group()
def cli():
    """Project Index - encuentra la mejor combinación de herramientas para tu tarea."""
    pass

@cli.command()
def init():
    """Inicializa el índice en ~/.project-index/"""
    INDEX_PATH.parent.mkdir(exist_ok=True)
    # Copia el índice por defecto
    console.print("[green]Index initialized at ~/.project-index/index.yaml[/green]")

@cli.command()
@click.argument('query')
def search(query):
    """Busca tareas que coincidan con la consulta."""
    index = load_index()
    if not index:
        return

    tasks = index.get('tasks', [])
    # Buscar coincidencias por descripción
    matches = process.extract(query, [t['description'] for t in tasks], limit=5)

    table = Table(title="Recomendaciones encontradas")
    table.add_column("ID", style="cyan")
    table.add_column("Área", style="green")
    table.add_column("Descripción")
    table.add_column("CLI", style="yellow")
    table.add_column("Modelo", style="magenta")

    for match in matches:
        desc = match[0]
        score = match[1]
        task = next(t for t in tasks if t['description'] == desc)
        rec = task['recommendations']
        table.add_row(
            task['id'],
            task['area'],
            desc[:50] + "...",
            rec.get('cli', ''),
            rec.get('model', '')
        )

    console.print(table)

if __name__ == '__main__':
    cli()

### 🧠 1. El Motor de Decisión: ¿Qué Afecta la Ejecución?

La adaptación de la herramienta no es una elección única, sino un equilibrio entre varias variables críticas. La base de datos debe modelar estas dimensiones para cada tarea:

*   **Rendimiento (Quality)**: La capacidad del modelo para resolver la tarea correctamente. Se mide con benchmarks como **SWE-bench** para codificación, **GPQA** para razonamiento, y benchmarks especializados como **DeepSWE** para tareas de ingeniería de software a largo plazo.
*   **Coste (Cost)**: El coste por millón de tokens (entrada/salida). Modelos como `GPT-5.6 Terra` ofrecen reducciones de coste del 60-80% en comparación con modelos más potentes.
*   **Velocidad (Speed/Latencia)**: El tiempo de respuesta, crucial para tareas interactivas o de desarrollo rápido.
*   **Contexto (Context Window)**: La cantidad de información que el modelo puede procesar de una vez. Crítico para proyectos grandes o análisis de código extenso.
*   **Seguridad (Security)**: La capacidad del modelo para generar código seguro y resistente a vulnerabilidades.
*   **Capacidad Agéntica (Agentic Capability)**: La habilidad para usar herramientas, planificar y ejecutar tareas complejas de forma autónoma.

Cada tarea en tu índice ponderará estas variables de forma diferente. Por ejemplo:

*   **Frontend/Animaciones (GSAP)**: Ponderación alta en **Rendimiento Creativo** y **Velocidad**, ponderación media en **Coste**.
*   **Backend/API REST**: Ponderación alta en **Rendimiento de Codificación** y **Seguridad**, ponderación media en **Coste**.
*   **Seguridad/Auditoría de Código**: Ponderación muy alta en **Seguridad** y **Rendimiento de Razonamiento**, ponderación baja en **Velocidad**.
*   **Tareas Simples/Generación de Documentación**: Ponderación alta en **Velocidad** y **Coste**, ponderación media en **Rendimiento**.

#### b) Mecanismo de Actualización Automática

Para que la base de datos se autocorrija, necesitas un script que:

1.  **Scrapee líderes de benchmarks** públicos como [llm-stats.com](https://llm-stats.com), [LiveBench](https://livebench.ai), o [Klu.ai](https://klu.ai).
2.  **Extraiga los datos** de rendimiento de cada modelo.
3.  **Actualice la base de datos** (`performance_db.yaml`) con los nuevos valores.
4.  **Recalcule las recomendaciones** para cada tarea en `index.yaml` basándose en los nuevos datos.

Este script podría ejecutarse diariamente o semanalmente (vía GitHub Actions), asegurando que tu índice esté siempre actualizado.

---

### 🔄 3. El Mecanismo de Autocorrección (El "Cerebro" Adaptativo)

La base de datos no solo almacena datos, sino que también **aprende de las interacciones de los usuarios**. Cada vez que un usuario usa `meta-skill search` y selecciona una opción, o ejecuta `--execute` con un modelo, el sistema registra:

*   **Éxito/Fracaso**: ¿El modelo resolvió la tarea correctamente? (Esto podría ser feedback implícito, como "el usuario no volvió a preguntar", o explícito, como una calificación).
*   **Tiempo de respuesta**: ¿Fue rápido?
*   **Coste real**: ¿Cuánto costó la ejecución?

Estos datos alimentan un **bucle de retroalimentación** que ajusta las ponderaciones de las variables para cada tarea. Por ejemplo, si los usuarios eligen constantemente `GPT-5.6 Sol` para tareas de seguridad y obtienen buenos resultados, el sistema aumentará la ponderación de `security_score` para esas tareas.

Este mecanismo se basa en técnicas de **enrutamiento dinámico** y **aprendizaje por refuerzo**, donde el sistema aprende a elegir el mejor modelo para cada situación sin necesidad de intervención manual constante.

---

### 🧩 4. La Matriz de Decisión Final (Cómo se Unifica Todo)

Para cada tarea en tu `index.yaml`, el sistema realiza los siguientes pasos:

1.  **Recupera la tarea** y sus ponderaciones de variables.
2.  **Consulta la base de datos de rendimiento** para obtener las puntuaciones de todos los modelos disponibles.
3.  **Calcula una puntuación compuesta** para cada modelo, basada en las ponderaciones de la tarea y las puntuaciones del modelo.
4.  **Selecciona el modelo con la puntuación más alta** y devuelve la recomendación completa (CLI, API, skill, etc.).
5.  **Registra la interacción** para futuras actualizaciones.

Este proceso convierte tu `Meta-Skill Index` en un sistema de recomendación inteligente y adaptativo, muy superior a un simple catálogo estático.

#### 🤖 `scripts/fetch_openrouter.py` (El Bot Zero-Token)
Este script consume la API gratuita de OpenRouter y genera `models.yaml` con las capacidades reales de los modelos.

```python
import requests
import yaml
import os

def fetch_and_score():
    print("🔄 Conectando con API pública de OpenRouter...")
    response = requests.get("https://openrouter.ai/api/v1/models", timeout=15)
    response.raise_for_status()
    raw_models = response.json().get('data', [])

    scored_models = []
    # Filtramos solo los modelos relevantes para desarrollo
    target_keywords = ["claude", "gpt", "gemini", "qwen", "deepseek", "llama", "mistral"]

    for m in raw_models:
        model_id = m.get('id', '')
        if not any(k in model_id.lower() for k in target_keywords):
            continue
            
        context = m.get('context_length', 4096)
        prompt_price = float(m.get('pricing', {}).get('prompt', '0'))
        
        # Heurística de Puntuación (0 a 100)
        scores = {
            "context": 100 if context >= 1000000 else (90 if context >= 200000 else 60),
            "cost": 100 if prompt_price == 0 else (80 if prompt_price < 1.0 else 40),
            "speed": 70, # Base
            "reasoning": 85 if "opus" in model_id or "pro" in model_id or "o1" in model_id else 70,
            "coding": 95 if "coder" in model_id or "claude" in model_id else 75,
            "security": 90 if "claude" in model_id else 70
        }

        scored_models.append({
            "id": model_id,
            "context_window": context,
            "prompt_price_per_1m": prompt_price,
            "scores": scores
        })

    # Guardar en data/models.yaml
    os.makedirs('data', exist_ok=True)
    with open('data/models.yaml', 'w', encoding='utf-8') as f:
        yaml.dump({"models": scored_models, "last_updated": "auto"}, f, allow_unicode=True)
    
    print(f"✅ {len(scored_models)} modelos procesados y guardados.")

if __name__ == "__main__":
    fetch_and_score()
```

#### 🔄 `sync.py` (El Sincronizador)
Descarga los YAMLs crudos desde GitHub. Tráfico infinito y gratuito.

```python
import requests
from pathlib import Path

REPO_RAW = "https://raw.githubusercontent.com/TU-USUARIO/meta-skill-db/main/data/" # <-- CAMBIA TU USUARIO
DB_DIR = Path.home() / '.meta-skill'
DB_DIR.mkdir(exist_ok=True)

def sync():
    print("🔄 Sincronizando con Git (Zero-Token)...")
    for file in ["tasks.yaml", "models.yaml"]:
        try:
            r = requests.get(f"{REPO_RAW}{file}", timeout=5)
            if r.status_code == 200:
                (DB_DIR / file).write_text(r.text, encoding='utf-8')
                print(f"  ✅ {file} actualizado.")
            else:
                print(f"  ⚠️ {file} no encontrado en el repo.")
        except Exception as e:
            print(f"  ❌ Error de red: {e}")

if __name__ == "__main__":
    sync()
```

#### 🧠 `router.py` (El Motor Matemático)
El corazón del proyecto. Cruza tu consulta con `tasks.yaml` (usando BM25) y luego cruza los pesos con `models.yaml` (usando Álgebra Lineal).

```python
import yaml
import numpy as np
import click
from pathlib import Path
from rich.console import Console
from rich.panel import Panel
from rank_bm25 import BM25Okapi
import nltk
from nltk.tokenize import word_tokenize
from nltk.corpus import stopwords

# Setup silencioso de NLTK
nltk.download('punkt', quiet=True)
nltk.download('stopwords', quiet=True)

console = Console()
DB_DIR = Path.home() / '.meta-skill'

def load_db():
    with open(DB_DIR / 'tasks.yaml', 'r', encoding='utf-8') as f: tasks = yaml.safe_load(f)['tasks']
    with open(DB_DIR / 'models.yaml', 'r', encoding='utf-8') as f: models = yaml.safe_load(f)['models']
    return tasks, models

def route(query):
    tasks, models = load_db()
    
    # 1. BM25: Encontrar la tarea (Léxico, Cero Tokens)
    corpus = [task['keywords'] for task in tasks]
    bm25 = BM25Okapi(corpus)
    
    stop_words = set(stopwords.words('spanish'))
    q_tokens = [w for w in word_tokenize(query.lower()) if w.isalnum() and w not in stop_words]
    
    scores = bm25.get_scores(q_tokens)
    best_idx = np.argmax(scores)
    
    if scores[best_idx] == 0:
        console.print("[yellow]⚠️ No se encontró ninguna tarea relacionada.[/yellow]")
        return

    task = tasks[best_idx]
    weights = task['weights']
    
    # 2. Álgebra Lineal: Encontrar el mejor modelo (Dot Product)
    best_model = None
    max_score = -1
    
    for model in models:
        m_scores = model.get('scores', {})
        # Producto Escalar: Sum(Weight[i] * Stat[i])
        score = sum(weights.get(k, 0) * m_scores.get(k, 50) for k in weights)
        if score > max_score:
            max_score = score
            best_model = model

    # 3. Output con System Prompt Comprimido (Ahorro de Tokens en Destino)
    panel = f"""
[bold cyan]🎯 Tarea Detectada:[/bold cyan] {task['id']} (Score BM25: {scores[best_idx]:.2f})

[bold green]🏆 Modelo Óptimo Matemático:[/bold green] {best_model['id']}
[yellow]↳ Ventana de Contexto:[/yellow] {best_model['context_window']:,} tokens
[yellow]↳ Coste Input:[/yellow] ${best_model['prompt_price_per_1m']}/1M tokens

[bold magenta]🧠 System Prompt Comprimido (Copia y pega esto en tu CLI para ahorrar tokens):[/bold magenta]
"{task['system_prompt']}"

[bold]💡 CLIs recomendados en tu máquina:[/bold] {', '.join(task.get('cli_fallback', []))}
    """
    console.print(Panel(panel, title="🧠 Meta-Skill Router (Git-Based)", border_style="green"))

@click.command()
@click.argument('query')
def main(query):
    route(query)

if __name__ == "__main__":
    main()
```
